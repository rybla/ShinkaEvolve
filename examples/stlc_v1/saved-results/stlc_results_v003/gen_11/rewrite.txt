from typing import Literal, Tuple, List
import random

"""
`Ty` and `Tm` define the shape of a simply-typed lambda-calculus deeply embedded in Python.
"""

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]


type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generate a sample type and term using a random seed, optimized for 
    variable usage, term size, and structural diversity.
    """

    def gen_type(budget: int) -> Ty:
        if budget <= 1 or rng.random() < 0.3:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        left_budget = rng.randint(1, budget - 1)
        return ("Fun", gen_type(left_budget), gen_type(budget - left_budget))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        # Identify variables in context that match the required type
        valid_vars = [name for name, ty in ctx if ty == target_ty]

        # Base cases: Use variables if possible, otherwise literals
        if budget <= 2:
            if valid_vars and (rng.random() < 0.9 or budget <= 1):
                return ("Var", rng.choice(valid_vars))
            
            if target_ty[0] == "Bool":
                return ("Bool", rng.choice([True, False]))
            if target_ty[0] == "Int":
                return ("Int", rng.randint(0, 100))
            if target_ty[0] == "String":
                return ("String", rng.choice(["lambda", "stlc", "type"]))
            # Fallback for Fun types if budget is too low for Lam/App
            if target_ty[0] == "Fun":
                # This case is rare due to recursive logic but ensures validity
                var_name = f"v{len(ctx)}"
                return ("Lam", var_name, target_ty[1], gen_term(target_ty[2], ctx + [(var_name, target_ty[1])], 1))

        # Recursive Case 1: Lambda Abstraction
        # If target is a function, prioritize generating a Lambda to increase depth
        if target_ty[0] == "Fun" and (rng.random() < 0.7 or budget < 5):
            arg_ty, ret_ty = target_ty[1], target_ty[2]
            var_name = f"v{len(ctx)}"
            new_ctx = ctx + [(var_name, arg_ty)]
            return ("Lam", var_name, arg_ty, gen_term(ret_ty, new_ctx, budget - 1))

        # Recursive Case 2: Application
        # To generate (App f x) : target_ty, we need f : A -> target_ty and x : A
        # Occasionally pick A from context to boost variable usage
        if ctx and rng.random() < 0.4:
            arg_ty = rng.choice(ctx)[1]
        else:
            arg_ty = gen_type(rng.randint(1, 2))
            
        fun_ty = ("Fun", arg_ty, target_ty)

        # Asymmetric split: allocate more budget to the function side (left)
        # to encourage deep application chains and higher applied_lambdas_proportions.
        split_f = int(budget * 0.7)
        split_x = budget - split_f - 1
        
        return ("App", 
                gen_term(fun_ty, ctx, max(1, split_f)), 
                gen_term(arg_ty, ctx, max(1, split_x)))

    # Target type size ~10 nodes, Term size ~100 nodes
    target_type = gen_type(10)
    # Start with a healthy budget to maximize AST size
    term = gen_term(target_type, [], 100)

    return target_type, term

# This part remains fixed (not evolved)

import lc


def run() -> lc.RunOutput:
    """Run the analysis"""

    size = 1000
    rng = random.Random()

    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
    )
