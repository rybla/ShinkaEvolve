I will implement a more sophisticated generator that tracks variable contexts and target types to ensure well-typedness while maximizing the size and complexity of the generated terms. 

Key improvements:
1. **Context Tracking**: Pass a `ctx` (mapping variable names to types) to allow the generation of `Var` terms.
2. **Targeted Generation**: The generator now takes a target `Ty` as an argument, ensuring the term matches the type.
3. **Size Management**: Uses a `budget` parameter to control the AST size for both types and terms, aiming for the 10-node limit for types and 100-node limit for terms.
4. **Higher-Order Terms**: Encourages the creation of `App` and `Lam` nodes to increase the `applied_lambdas` and `used_vars` metrics.

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generate a sample type and term using a random seed.
    """

    def gen_type(budget: int) -> Ty:
        if budget <= 1:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        if rng.random() < 0.3:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        left_budget = rng.randint(1, budget - 1)
        return ("Fun", gen_type(left_budget), gen_type(budget - left_budget))

    def gen_term(target_ty: Ty, ctx: list[Tuple[str, Ty]], budget: int) -> Tm:
        # Try to use a variable from context if it matches the type
        valid_vars = [name for name, ty in ctx if ty == target_ty]
        
        # Base cases or if budget is low
        if budget <= 1 or (not target_ty[0] == "Fun" and rng.random() < 0.2):
            if valid_vars and (rng.random() < 0.7 or budget <= 1):
                return ("Var", rng.choice(valid_vars))
            if target_ty == ("Bool",):
                return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",):
                return ("Int", rng.randint(0, 100))
            if target_ty == ("String",):
                return ("String", rng.choice(["a", "b", "c"]))

        # Recursive cases
        if target_ty[0] == "Fun" and rng.random() < 0.6:
            # Generate Lambda
            arg_ty, ret_ty = target_ty[1], target_ty[2]
            var_name = f"v{len(ctx)}"
            new_ctx = ctx + [(var_name, arg_ty)]
            return ("Lam", var_name, arg_ty, gen_term(ret_ty, new_ctx, budget - 1))
        
        # Generate Application
        # To generate (App f x) : target_ty, we need f : A -> target_ty and x : A
        arg_ty = gen_type(rng.randint(1, 3))
        fun_ty = ("Fun", arg_ty, target_ty)
        
        split = max(1, budget // 2)
        return ("App", gen_term(fun_ty, ctx, split), gen_term(arg_ty, ctx, split))

    # Target type size ~10 nodes, Term size ~100 nodes
    target_type = gen_type(10)
    term = gen_term(target_type, [], 100)
    
    return target_type, term
