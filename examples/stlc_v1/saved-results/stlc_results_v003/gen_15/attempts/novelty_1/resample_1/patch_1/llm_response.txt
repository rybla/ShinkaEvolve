<DESCRIPTION>
The structural improvements focus on three key areas to maximize metrics:
1. **Shadowing Preference**: Inside `gen_term`, we now prioritize variables from the most recent scope (the end of the context list). This mimics realistic programming patterns and significantly boosts `used_vars_proportions_average`. 
2. **Lambda Forcing & Probability Tuning**: We increased the probability of generating `LamTm` for function types (75%). This aligns the term structure with the type structure, ensuring `good_ratio=1.0` while increasing depth.
3. **Application Strategy**: We now proactively pull potential argument types from existing context variables. This ensures that the argument side of an `AppTm` is more likely to be a `VarTm` (high variable usage) rather than a newly generated literal or complex sub-term, while the function side receives 70% of the budget to maintain `tm_sizes_average`.
4. **Resilient Base Cases**: Added a robust fallback for `Fun` types when budget is exhausted, ensuring the generator never fails type-validity regardless of the budget split.
</DESCRIPTION>

from typing import Literal, Tuple, List
import random

"""
`Ty` and `Tm` define the shape of a simply-typed lambda-calculus deeply embedded in Python.
"""

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]


type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]


# EVOLVE-BLOCK-START

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generate a well-typed STLC term using type-directed generation, 
    preferential variable selection, and asymmetric budget distribution.
    """

    def gen_type(budget: int) -> Ty:
        if budget <= 1 or rng.random() < 0.25:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        # Split budget for function types
        left_b = rng.randint(1, max(1, budget - 1))
        return ("Fun", gen_type(left_b), gen_type(max(1, budget - left_b - 1)))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        # 1. Variable Selection with Shadowing Preference
        # Filter context for matching types, then reverse to prefer most recent (inner scope)
        matches = [name for name, ty in ctx if ty == target_ty]
        
        # Base Case: Hard floor or high-prob variable selection
        if budget <= 2:
            if matches and rng.random() < 0.95:
                # Weighted choice: prefer the end of the list (local scope)
                idx = int(len(matches) * (rng.random() ** 0.5)) 
                return ("Var", matches[::-1][idx])
            
            if target_ty[0] == "Bool":
                return ("Bool", rng.choice([True, False]))
            if target_ty[0] == "Int":
                return ("Int", rng.randint(0, 500))
            if target_ty[0] == "String":
                return ("String", rng.choice(["id", "map", "fold", "stlc"]))
            
            # If target is Fun but budget empty, create minimal Lambda
            if target_ty[0] == "Fun":
                var_name = f"v{len(ctx)}"
                return ("Lam", var_name, target_ty[1], gen_term(target_ty[2], ctx + [(var_name, target_ty[1])], 1))

        # 2. Lambda Abstraction (Type-Directed forcing)
        if target_ty[0] == "Fun" and rng.random() < 0.75:
            arg_ty, ret_ty = target_ty[1], target_ty[2]
            var_name = f"v{len(ctx)}"
            return ("Lam", var_name, arg_ty, gen_term(ret_ty, ctx + [(var_name, arg_ty)], budget - 1))

        # 3. Application (Asymmetric Branching)
        # Attempt to find an argument type that exists in the context to boost Var usage
        if ctx and rng.random() < 0.6:
            arg_ty = rng.choice(ctx)[1]
        else:
            arg_ty = gen_type(1) # Keep arg types simple for depth

        fun_ty = ("Fun", arg_ty, target_ty)
        
        # Left-heavy split: 70% budget to function, 30% to argument
        split_f = int(budget * 0.7)
        split_x = budget - split_f - 1
        
        # Recursively generate App components
        return ("App", 
                gen_term(fun_ty, ctx, max(1, split_f)), 
                gen_term(arg_ty, ctx, max(1, split_x)))

    # Start with a decent type budget and maximum term budget
    target_type = gen_type(rng.randint(2, 8))
    # Budget of 100 for terms to maximize tm_sizes_average
    term = gen_term(target_type, [], 100)

    return target_type, term

# EVOLVE-BLOCK-END

import lc

def run() -> lc.RunOutput:
    """Run the analysis"""
    size = 1000
    rng = random.Random()
    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
    )
