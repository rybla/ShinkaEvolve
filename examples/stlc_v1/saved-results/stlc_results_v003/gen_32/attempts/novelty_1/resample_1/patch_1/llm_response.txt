<DESCRIPTION>
This generator combines the high-efficiency components of previous models to maximize structural complexity and variable reuse. 
1. **Saturation Budgeting**: Uses a high initial fuel (280) to consistently reach the 100-node limit.
2. **Context-Length Weighted Lambda Forcing**: Employs a high early-stage abstraction probability (98% for small contexts) to build a rich environment of variables, then scales back to 85% to allow for deep application spines.
3. **Optimized Variable Reuse**: Uses frequency-weighted selection from the context when generating applications, ensuring that the argument branch is highly likely to contain a valid variable reference.
4. **Asymmetric Spine Split**: Implements a 92/8 split for `AppTm` nodes, creating deep, left-leaning trees that maximize unique sub-term counts and total size.
5. **Hard-Shadowing Selection**: Always selects the most recent variable from the context (`matches[-1]`) to maximize local binding usage.

from typing import Literal, Tuple, List
import random
import tqdm

"""
`Ty` and `Tm` define the shape of a simply-typed lambda-calculus deeply embedded in Python.
"""

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]


type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    STLC generator optimized for node count, variable reuse, and applied lambdas.
    Uses asymmetric recursion and popularity-based context selection.
    """

    def gen_type(budget: int) -> Ty:
        if budget <= 1 or rng.random() < 0.25:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        b_left = rng.randint(1, max(1, budget - 1))
        return ("Fun", gen_type(b_left), gen_type(max(1, budget - b_left - 1)))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        # 1. Base Case / Variable Selection
        matches = [name for name, ty in ctx if ty == target_ty]

        if budget <= 3:
            if matches and rng.random() < 0.98:
                # Prioritize the most recently added matching variable (local shadowing)
                return ("Var", matches[-1])
            
            # Forced Literals for Base Cases
            if target_ty[0] == "Bool": return ("Bool", rng.choice([True, False]))
            if target_ty[0] == "Int": return ("Int", rng.randint(0, 1000))
            if target_ty[0] == "String": return ("String", rng.choice(["val", "stlc", "x"]))
            
            # Valid inhabitant for Fun even if budget is 0
            if target_ty[0] == "Fun":
                v_name = f"v{len(ctx)}"
                return ("Lam", v_name, target_ty[1], gen_term(target_ty[2], ctx + [(v_name, target_ty[1])], 1))

        # 2. Forced Lambda Promotion
        # Increase applied_lambdas_proportions by ensuring Fun types are usually Lam
        lambda_prob = 0.98 if len(ctx) < 4 else 0.85
        if target_ty[0] == "Fun" and rng.random() < lambda_prob:
            arg_ty, ret_ty = target_ty[1], target_ty[2]
            v_name = f"v{len(ctx)}"
            return ("Lam", v_name, arg_ty, gen_term(ret_ty, ctx + [(v_name, arg_ty)], budget - 1))

        # 3. Application Strategy (Context-Aware)
        # 90% chance to pick an arg_ty from the existing context to maximize Variable use
        if ctx and rng.random() < 0.90:
            # Random pick from context automatically weights by type frequency
            arg_ty = rng.choice(ctx)[1]
        else:
            arg_ty = gen_type(1)

        fun_ty = ("Fun", arg_ty, target_ty)
        
        # Asymmetric Split (92/8) to create massive left-heavy spines
        # Fuel is consumed primarily by the functional part of the application
        split_f = int(budget * 0.92)
        split_x = budget - split_f - 1
        
        return ("App", 
                gen_term(fun_ty, ctx, max(1, split_f)), 
                gen_term(arg_ty, ctx, max(1, split_x)))

    # Start with a robust type complexity
    target_type = gen_type(rng.randint(6, 12))
    
    # Use a large budget to saturate the 100 node limit
    term = gen_term(target_type, [], 280)

    return target_type, term

def run() -> lc.RunOutput:
    """Run the analysis"""

    size = 1000
    rng = random.Random()

    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
