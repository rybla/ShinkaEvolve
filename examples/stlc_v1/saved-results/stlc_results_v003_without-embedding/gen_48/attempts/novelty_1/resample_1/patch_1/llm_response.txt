def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generate a sample type and term using a random seed.
    """

    def gen_type(budget: int) -> Ty:
        if budget <= 1:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        if rng.random() < 0.3:
            return rng.choice([("Bool",), ("Int",), ("String",)])

        left_budget = rng.randint(1, budget - 1)
        return ("Fun", gen_type(left_budget), gen_type(budget - left_budget))

    def gen_term(target_ty: Ty, ctx: list[Tuple[str, Ty]], budget: int) -> Tm:
        # Try to use a variable from context if it matches the type
        valid_vars = [name for name, ty in ctx if ty == target_ty]

        # Base cases or if budget is low
        if budget <= 1 or (not target_ty[0] == "Fun" and rng.random() < 0.2):
            if valid_vars and (rng.random() < 0.7 or budget <= 1):
                return ("Var", rng.choice(valid_vars))
            if target_ty == ("Bool",):
                return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",):
                return ("Int", rng.randint(0, 100))
            if target_ty == ("String",):
                return ("String", rng.choice(["a", "b", "c"]))

        # Recursive cases
        if target_ty[0] == "Fun" and rng.random() < 0.6:
            # Generate Lambda
            arg_ty, ret_ty = target_ty[1], target_ty[2]
            var_name = f"v{len(ctx)}"
            new_ctx = ctx + [(var_name, arg_ty)]
            return ("Lam", var_name, arg_ty, gen_term(ret_ty, new_ctx, budget - 1))

        # Generate Application
        # To generate (App f x) : target_ty, we need f : A -> target_ty and x : A
        arg_ty = gen_type(rng.randint(1, 3))
        fun_ty = ("Fun", arg_ty, target_ty)

        split = max(1, budget // 2)
        return ("App", gen_term(fun_ty, ctx, split), gen_term(arg_ty, ctx, split))

    # Target type size ~10 nodes, Term size ~100 nodes
    target_type = gen_type(10)
    term = gen_term(target_type, [], 100)

    return target_type, term

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generates STLC terms with extreme spine depth (95/5 budget),
    context-aware variable reuse, and manual beta-redex injection.
    """

    def gen_type(budget: int) -> Ty:
        if budget <= 1 or rng.random() < 0.2:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        left_b = rng.randint(1, max(1, budget - 1))
        return ("Fun", gen_type(left_b), gen_type(max(1, budget - left_b)))

    def gen_term(target_ty: Ty, ctx: list[Tuple[str, Ty]], budget: int) -> Tm:
        matches = [n for n, t in ctx if t == target_ty]

        # 1. Hard Floor / Terminal Case (Budget <= 2)
        if budget <= 2:
            if matches and rng.random() < 0.98:
                return ("Var", matches[-1]) # Shadowing bias
            if target_ty[0] == "Fun":
                v = f"v{len(ctx)}_{rng.getrandbits(16)}"
                return ("Lam", v, target_ty[1], gen_term(target_ty[2], ctx + [(v, target_ty[1])], 1))
            if target_ty[0] == "Bool": return ("Bool", rng.choice([True, False]))
            if target_ty[0] == "Int": return ("Int", rng.getrandbits(30))
            return ("String", f"s_{rng.getrandbits(30)}")

        # 2. Functional Abstraction
        if target_ty[0] == "Fun" and rng.random() < 0.88:
            v = f"v{len(ctx)}_{rng.getrandbits(16)}"
            return ("Lam", v, target_ty[1], gen_term(target_ty[2], ctx + [(v, target_ty[1])], budget - 1))

        # 3. Application Branch
        # Pick arg_ty from context to ensure VarTm can satisfy the argument branch
        if ctx and rng.random() < 0.95:
            arg_ty = rng.choice(ctx)[1]
        else:
            arg_ty = gen_type(1)
        
        fun_ty = ("Fun", arg_ty, target_ty)
        # Apply 95/5 Asymmetric Budgeting
        split_f = max(1, int(budget * 0.95))
        split_a = budget - split_f - 1

        # Manual Beta-Redex Injection
        if budget > 50 and rng.random() < 0.35:
            v = f"v{len(ctx)}_{rng.getrandbits(16)}"
            fun_tm = ("Lam", v, arg_ty, gen_term(target_ty, ctx + [(v, arg_ty)], split_f - 1))
            return ("App", fun_tm, gen_term(arg_ty, ctx, max(1, split_a)))

        return ("App", gen_term(fun_ty, ctx, split_f), gen_term(arg_ty, ctx, max(1, split_a)))

    # Use a small target type to maximize budget for terms
    target_type = gen_type(rng.randint(2, 4))
    # Fuel of 500 pushes term_sizes_average significantly higher
    term = gen_term(target_type, [], 500)

    return target_type, term
