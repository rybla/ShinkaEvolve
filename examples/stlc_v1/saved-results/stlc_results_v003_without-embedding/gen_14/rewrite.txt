from typing import Literal, Tuple, List
import random

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]

type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

# EVOLVE-BLOCK-START

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generates well-typed STLC terms with high variable density and deep ASTs
    using asymmetric recursion and context-aware type selection.
    """

    def gen_type(budget: int) -> Ty:
        if budget <= 1 or rng.random() < 0.3:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        # Split budget for function types
        left_b = rng.randint(1, budget - 1)
        return ("Fun", gen_type(left_b), gen_type(budget - left_b))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        # 1. Identify matching variables in the current context
        valid_vars = [name for name, ty in ctx if ty == target_ty]

        # 2. Base Case: Budget exhausted or forced leaf
        if budget <= 2:
            if valid_vars and (rng.random() < 0.95 or budget <= 1):
                return ("Var", rng.choice(valid_vars))
            
            # Literal Fallbacks
            if target_ty[0] == "Bool":
                return ("Bool", rng.choice([True, False]))
            if target_ty[0] == "Int":
                return ("Int", rng.randint(0, 100))
            if target_ty[0] == "String":
                return ("String", rng.choice(["lambda", "stlc", "type", "eval"]))
            
            # If target is Fun but budget is dead, we must still return a valid term.
            # We force a small Lambda to maintain good_ratio = 1.0.
            if target_ty[0] == "Fun":
                var_name = f"v{len(ctx)}"
                return ("Lam", var_name, target_ty[1], gen_term(target_ty[2], ctx + [(var_name, target_ty[1])], 1))

        # 3. Lambda Generation (Forcing structure for Function types)
        if target_ty[0] == "Fun" and (rng.random() < 0.85):
            arg_ty, ret_ty = target_ty[1], target_ty[2]
            var_name = f"v{len(ctx)}"
            return ("Lam", var_name, arg_ty, gen_term(ret_ty, ctx + [(var_name, arg_ty)], budget - 1))

        # 4. Application Generation (Asymmetric Splitting)
        # Strategy: Pick an intermediate type 'A' from context to increase Var usage
        if ctx and rng.random() < 0.6:
            arg_ty = rng.choice(ctx)[1]
        else:
            # Otherwise generate a very simple type
            arg_ty = gen_type(1)
            
        fun_ty = ("Fun", arg_ty, target_ty)

        # Asymmetric split: 75% to the function (left), 25% to the argument (right)
        # This creates deep "spine" structures rather than wide trees.
        split_f = int(budget * 0.75)
        split_x = budget - split_f - 1
        
        return ("App", 
                gen_term(fun_ty, ctx, max(1, split_f)), 
                gen_term(arg_ty, ctx, max(1, split_x)))

    # Initial generation parameters
    # Type budget 8-12, Term budget 100-120
    target_type = gen_type(rng.randint(8, 12))
    term = gen_term(target_type, [], 110)

    return target_type, term

# EVOLVE-BLOCK-END

import lc

def run() -> lc.RunOutput:
    """Run the analysis"""
    size = 1000
    rng = random.Random()
    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
    )
