from typing import Literal, Tuple, List
import random

"""
`Ty` and `Tm` define the shape of a simply-typed lambda-calculus deeply embedded in Python.
"""

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]


type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    STLC Generator optimized for extreme term depth, variable saturation, 
    and sub-term uniqueness through dynamic literal generation.
    """

    def gen_type(budget: int) -> Ty:
        if budget <= 1 or rng.random() < 0.25:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        b_left = rng.randint(1, max(1, budget - 1))
        return ("Fun", gen_type(b_left), gen_type(max(1, budget - b_left - 1)))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        # Pre-filter matching variables for the current type
        matches = [name for name, ty in ctx if ty == target_ty]

        # 1. Hard Floor Logic (Base Cases)
        if budget <= 3:
            if matches and rng.random() < 0.98:
                # Shadowing preference: use the most recent variable
                return ("Var", matches[-1])
            
            # Dynamic Literals to ensure unique_subterms_proportions remains high
            if target_ty[0] == "Bool":
                return ("Bool", rng.random() > 0.5)
            if target_ty[0] == "Int":
                # Entropy based on local state
                return ("Int", rng.randint(0, 10000) + len(ctx))
            if target_ty[0] == "String":
                # Ensure identity is unique via budget/context strings
                return ("String", f"v_{len(ctx)}_{budget}_{rng.randint(0, 100)}")
            
            if target_ty[0] == "Fun":
                v_name = f"x{len(ctx)}"
                return ("Lam", v_name, target_ty[1], gen_term(target_ty[2], ctx + [(v_name, target_ty[1])], 1))

        # 2. Forced Abstraction (Lambda Density)
        # Maintaining high applied_lambdas_proportions_average
        if target_ty[0] == "Fun" and rng.random() < 0.88:
            arg_ty, ret_ty = target_ty[1], target_ty[2]
            v_name = f"x{len(ctx)}"
            return ("Lam", v_name, arg_ty, gen_term(ret_ty, ctx + [(v_name, arg_ty)], budget - 1))

        # 3. Application Strategy (Context-Aware & Asymmetric)
        # High probability to pick a type already in context to maximize Var usage
        if ctx and rng.random() < 0.90:
            arg_ty = rng.choice([t for _, t in ctx])
        else:
            arg_ty = gen_type(1)

        fun_ty = ("Fun", arg_ty, target_ty)
        
        # 92/8 Asymmetric Split for extremely deep spines
        split_f = int(budget * 0.92)
        split_x = budget - split_f - 1
        
        return ("App", 
                gen_term(fun_ty, ctx, max(1, split_f)), 
                gen_term(arg_ty, ctx, max(1, split_x)))

    # Type complexity 8-10 nodes
    target_type = gen_type(rng.randint(8, 10))
    
    # Large budget to drive tm_sizes_average towards 100
    # The generator handles budget exhaustion gracefully via the hard floor.
    term = gen_term(target_type, [], 350)

    return target_type, term

# This part remains fixed (not evolved)

import lc


def run() -> lc.RunOutput:
    """Run the analysis"""

    size = 1000
    rng = random.Random()

    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
    )
