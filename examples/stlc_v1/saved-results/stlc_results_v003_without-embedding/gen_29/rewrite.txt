from typing import Literal, Tuple, List
import random

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]

type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

# EVOLVE-BLOCK-START

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generates well-typed STLC terms using asymmetric recursion and 
    context-driven type selection to maximize term depth and variable reuse.
    """

    def gen_type(budget: int) -> Ty:
        if budget <= 1 or rng.random() < 0.25:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        b_left = rng.randint(1, max(1, budget - 1))
        return ("Fun", gen_type(b_left), gen_type(max(1, budget - b_left - 1)))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        # 1. Variable Selection (Maximize used_vars_proportions_average)
        matches = [name for name, ty in ctx if ty == target_ty]

        # Hard Floor Base Case
        if budget <= 2:
            if matches and rng.random() < 0.95:
                # Prefer most recent variable (shadowing)
                return ("Var", matches[-1])
            
            if target_ty[0] == "Bool":
                return ("Bool", rng.choice([True, False]))
            if target_ty[0] == "Int":
                return ("Int", rng.randint(0, 1000))
            if target_ty[0] == "String":
                return ("String", rng.choice(["a", "b", "c"]))
            
            if target_ty[0] == "Fun":
                v_name = f"v{len(ctx)}"
                return ("Lam", v_name, target_ty[1], gen_term(target_ty[2], ctx + [(v_name, target_ty[1])], 1))

        # 2. Context-Length Dependent Lambda Forcing
        # Aggressively build context early (98%), then stabilize (75%)
        lambda_prob = 0.98 if len(ctx) < 5 else 0.75
        if target_ty[0] == "Fun" and rng.random() < lambda_prob:
            arg_ty, ret_ty = target_ty[1], target_ty[2]
            v_name = f"v{len(ctx)}"
            return ("Lam", v_name, arg_ty, gen_term(ret_ty, ctx + [(v_name, arg_ty)], budget - 1))

        # 3. Application Strategy (92/8 Asymmetric Spine)
        # Pick an argument type from the context to increase variable hits
        if ctx and rng.random() < 0.90:
            arg_ty = rng.choice(ctx)[1]
        else:
            arg_ty = gen_type(1)

        fun_ty = ("Fun", arg_ty, target_ty)
        
        # Extremely left-heavy split to maximize tm_sizes_average
        split_f = int(budget * 0.92)
        split_x = budget - split_f - 1
        
        return ("App", 
                gen_term(fun_ty, ctx, max(1, split_f)), 
                gen_term(arg_ty, ctx, max(1, split_x)))

    # Target type size 6-12 for sufficient functional complexity
    target_type = gen_type(rng.randint(6, 12))
    
    # Large starting budget (250) to ensure average size > 60 after recursion pruning
    term = gen_term(target_type, [], 250)

    return target_type, term

# EVOLVE-BLOCK-END

import lc

def run() -> lc.RunOutput:
    """Run the analysis"""
    size = 1000
    rng = random.Random()
    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
    )
