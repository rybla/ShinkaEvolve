<DESCRIPTION>
The new algorithm focuses on increasing 'applied_lambdas' and 'used_vars' by introducing "Higher-Order Redexes". Instead of passing simple variables or literals into redexes, it biases the argument generation to produce function-typed arguments 50% of the time. It also introduces a "Deep Currying" mechanism that searches the context for variables capable of reaching the target type through multiple applications, and a "Redex-Chain" that generates nested abstractions applied to arguments. The budget allocation is tuned to favor the operator side (70%) of applications to allow for deeper nested structures while remaining under the 100-node limit.
</DESCRIPTION>

from typing import Literal, Tuple
import random

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]


type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generate a sample type and term using a random seed with a focus on 
    higher-order redexes and deep currying.
    """

    def gen_type(max_nodes: int) -> Ty:
        if max_nodes <= 1:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        if rng.random() < 0.75:  # High probability of function types
            left_size = rng.randint(1, max_nodes - 1)
            right_size = max_nodes - 1 - left_size
            return ("Fun", gen_type(left_size), gen_type(right_size))
        return rng.choice([("Bool",), ("Int",), ("String",)])

    def gen_term(ctx: list[Tuple[str, Ty]], target_ty: Ty, budget: int) -> Tm:
        # 1. Variable Match (Immediate satisfaction)
        matching_vars = [name for name, ty in ctx if ty == target_ty]
        if matching_vars and (budget <= 2 or rng.random() < 0.35):
            return ("Var", rng.choice(matching_vars))

        # 2. Deep Currying: Find context variables that return target_ty after N applications
        if budget > 8 and ctx:
            potential_callables = []
            for name, ty in ctx:
                args = []
                curr = ty
                while curr[0] == "Fun":
                    args.append(curr[1])
                    curr = curr[2]
                    if curr == target_ty:
                        potential_callables.append((name, args))
                        break
            
            if potential_callables and rng.random() < 0.5:
                vname, args_needed = rng.choice(potential_callables)
                acc = ("Var", vname)
                # Distribute budget across arguments
                arg_budget = max(1, budget // (len(args_needed) + 1))
                for arg_ty in args_needed:
                    acc = ("App", acc, gen_term(ctx, arg_ty, arg_budget))
                return acc

        # 3. Redex-Chain: ((\x.\y. body) arg1) arg2
        if budget > 30 and rng.random() < 0.25:
            # Force one argument to be a function type 50% of the time to boost complexity
            a1_ty = gen_type(3) if rng.random() < 0.5 else (rng.choice(ctx)[1] if ctx else gen_type(1))
            a2_ty = rng.choice(ctx)[1] if (ctx and rng.random() < 0.6) else gen_type(1)
            v1, v2 = f"v{len(ctx)}", f"v{len(ctx)+1}"
            
            body = gen_term(ctx + [(v1, a1_ty), (v2, a2_ty)], target_ty, int(budget * 0.5))
            inner_lam = ("Lam", v2, a2_ty, body)
            outer_lam = ("Lam", v1, a1_ty, inner_lam)
            
            return ("App", 
                    ("App", outer_lam, gen_term(ctx, a1_ty, budget // 5)), 
                    gen_term(ctx, a2_ty, budget // 5))

        # 4. Explicit Redex: (Lam x. body) arg
        if budget > 12 and rng.random() < 0.3:
            arg_ty = gen_type(2) if rng.random() < 0.5 else (rng.choice(ctx)[1] if ctx else gen_type(1))
            vname = f"v{len(ctx)}"
            lam = ("Lam", vname, arg_ty, gen_term(ctx + [(vname, arg_ty)], target_ty, int(budget * 0.6)))
            arg = gen_term(ctx, arg_ty, int(budget * 0.25))
            return ("App", lam, arg)

        # 5. Mandatory Lambda for Function Targets
        if target_ty[0] == "Fun":
            _, arg_ty, ret_ty = target_ty
            var_name = f"v{len(ctx)}"
            return ("Lam", var_name, arg_ty, gen_term(ctx + [(var_name, arg_ty)], ret_ty, budget - 1))

        # 6. General Application
        if budget > 3:
            # Bias arg_ty to context to increase used_vars
            if ctx and rng.random() < 0.8:
                arg_ty = rng.choice(ctx)[1]
            else:
                arg_ty = gen_type(rng.randint(1, 2))
            
            fun_ty = ("Fun", arg_ty, target_ty)
            # Favor the operator side to encourage nested Apps/Lams
            return ("App", 
                    gen_term(ctx, fun_ty, int(budget * 0.7)), 
                    gen_term(ctx, arg_ty, int(budget * 0.2)))

        # 7. Literals / Base Case
        if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
        if target_ty == ("Int",): return ("Int", rng.randint(0, 100))
        if target_ty == ("String",): return ("String", "s")

        return ("Int", 0) # Should be unreachable

    # Start with a decent type size and budget
    target_type = gen_type(rng.randint(6, 10))
    # Budget of 115 allows for complex terms while staying under the 100-node hard limit
    generated_term = gen_term([], target_type, 115)
    return target_type, generated_term


# EVOLVE-BLOCK-END

import lc


def run() -> lc.RunOutput:
    """Run the analysis"""

    size = 1000
    rng = random.Random()

    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
    )
