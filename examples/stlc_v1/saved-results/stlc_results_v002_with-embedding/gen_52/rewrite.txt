def generate_sample(rng: random.Random, depth=10, context=None) -> Tuple[Ty, Tm]:
    if context is None:
        context = []
    
    # At depth 0, only generate literals (3 options)
    # At depth > 0, generate literals, lambda, or application (5 options)
    if depth == 0:
        i = rng.randrange(0, 3)
    else:
        i = rng.randrange(0, 5)
    
    if i == 0:
        return (("Bool",), ("Bool", True))
    elif i == 1:
        return (("Int",), ("Int", 27))
    elif i == 2:
        return (("String",), ("String", "hello world"))
    elif i == 3:
        # Generate lambda abstraction with proper context update
        new_context = context + ["x"]
        ty, tm = generate_sample(rng, depth - 1, new_context)
        return (
            ("Fun", ("Int",), ty),
            ("Lam", "x", ("Int",), tm),
        )
    else:  # i == 4
        # Generate application - apply a lambda to an argument
        # First generate the function (lambda)
        arg_ty, arg_tm = generate_sample(rng, depth - 1, context)
        # Then generate the argument
        func_ty, func_tm = generate_sample(rng, depth - 1, context)
        return (
            ("App", func_ty, arg_ty),
            ("App", func_tm, arg_tm),
        )
