def generate_sample(rng: random.Random, depth=10) -> Tuple[Ty, Tm]:
    i = rng.randrange(0, 4 if depth > 0 else 3)
    
    if i == 0:
        return (("Bool",), ("Bool", True))
    elif i == 1:
        return (("Int",), ("Int", 27))
    elif i == 2:
        return (("String",), ("String", "hello world"))
    else:
        ty, tm = generate_sample(rng)
        return (
            ("Fun", ("Int",), ty),
            ("Lam", "x", ("Int",), tm),
        )
