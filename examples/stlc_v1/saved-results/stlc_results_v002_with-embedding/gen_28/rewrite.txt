def generate_sample(rng: random.Random, depth=10, context=None) -> Tuple[Ty, Tm]:
    if context is None:
        context = []
    
    i = rng.randrange(0, 4 if depth > 0 else 3)
    
    if i == 0:
        return (("Bool",), ("Bool", True))
    elif i == 1:
        return (("Int",), ("Int", 27))
    elif i == 2:
        return (("String",), ("String", "hello world"))
    else:
        if context and rng.random() < 0.3 and depth > 0:
            # Use a variable from context
            var_name = rng.choice(context)
            return (
                ("Fun", ("Int",), ("Int",)),
                ("Lam", "x", ("Int",), ("Var", var_name)),
            )
        else:
            # Add parameter to context when recursing
            new_context = context + ["x"]
            ty, tm = generate_sample(rng, depth - 1, new_context)
            return (
                ("Fun", ("Int",), ty),
                ("Lam", "x", ("Int",), tm),
            )
