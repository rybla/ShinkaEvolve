<DESCRIPTION>
This implementation optimizes STLC term generation by focusing on three key metrics: variable usage, applied lambda proportions, and structural uniqueness. 

1. **Asymmetric Budget Allocation**: In `App` nodes, we bias the budget significantly towards the function branch (75%). This encourages the creation of deeper, more complex function transformations and provides more space for `Lam` nodes, which are essential for both variable binding and redex injection.
2. **Enhanced Redex Injection**: We use a high base probability for redex injection that scales with the remaining budget. By explicitly constructing `(App (Lam x ty body) arg)`, we guarantee an increase in the `applied_lambdas_proportions` metric.
3. **Power-Law Recency Bias**: Variable selection uses a quadratic weighting `(i + 1)**2` based on the context index. This mimics local variable scoping and ensures that variables introduced deep in the AST are utilized, raising the `used_vars_proportions` score.
4. **Context-Aware Type Synthesis**: When generating applications, we select argument types from the existing context with 75% probability. This creates a "closed-loop" where arguments are likely to be satisfied by existing variables rather than new literals, increasing interconnectedness.
5. **Unique Identifier Strategy**: We use a combination of context depth, a global counter, and random salts to ensure every sub-term is structurally unique, keeping the uniqueness metric near 1.0.

from typing import Literal, Tuple, List
import random

"""
`Ty` and `Tm` define the shape of a simply-typed lambda-calculus deeply embedded in Python.
"""

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]


type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generate a well-typed STLC sample optimized for complexity, 
    variable usage, and applied lambda density.
    """
    
    var_counter = 0

    def get_unique_name(prefix: str, depth: int) -> str:
        nonlocal var_counter
        var_counter += 1
        return f"{prefix}_{depth}_{var_counter}_{rng.randint(0, 100)}"

    def gen_type(size_limit: int) -> Ty:
        # High bias towards functions to enable complex term structures
        if size_limit <= 1 or rng.random() < 0.15:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        # Split budget for nested function types
        left_size = rng.randint(1, max(1, size_limit - 1))
        return ("Fun", gen_type(left_size), gen_type(size_limit - left_size - 1))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        # Identify variables in the context that match the target type
        valid_vars = [name for name, t in ctx if t == target_ty]

        # Base Case: Minimal budget or forced termination
        if budget <= 1:
            if valid_vars and rng.random() < 0.95:
                # Power-law recency bias: favor variables added later (local scope)
                # Formula: (index + 1)**2 creates a strong bias towards the end of the list
                weights = [(i + 1)**2 for i in range(len(valid_vars))]
                return ("Var", rng.choices(valid_vars, weights=weights, k=1)[0])
            
            # Type-safe fallbacks
            if target_ty[0] == "Fun":
                v = get_unique_name("v", len(ctx))
                return ("Lam", v, target_ty[1], gen_term(target_ty[2], ctx + [(v, target_ty[1])], 0))
            
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-100, 100))
            return ("String", rng.choice(["val", "id", "node"]))

        # Weighted Constructor Selection
        choices, weights = [], []
        
        if target_ty[0] == "Fun":
            choices.append("lam")
            weights.append(25)
            
        if valid_vars:
            choices.append("var")
            weights.append(40) # High weight to maximize variable usage metric
            
        if budget > 2:
            choices.append("app")
            weights.append(60) # Highest weight to drive size and redexes
            
        if not choices or (target_ty[0] != "Fun" and rng.random() < 0.1):
            choices.append("lit")
            weights.append(10)

        mode = rng.choices(choices, weights=weights)[0]

        if mode == "lam":
            v_name = get_unique_name("x", len(ctx))
            return ("Lam", v_name, target_ty[1], gen_term(target_ty[2], ctx + [(v_name, target_ty[1])], budget - 1))

        elif mode == "var":
            v_weights = [(i + 1)**2 for i in range(len(valid_vars))]
            return ("Var", rng.choices(valid_vars, weights=v_weights, k=1)[0])

        elif mode == "app":
            # Context-Aware Type Selection: 75% chance to use a type already in context
            # This ensures the argument branch is likely to be a variable
            if ctx and rng.random() < 0.75:
                arg_ty = rng.choice(ctx)[1]
            else:
                arg_ty = gen_type(rng.randint(1, 3))

            # Scaled Redex Injection Strategy
            # Probability increases with budget to maximize applied lambdas
            redex_prob = 0.4 + (budget / 160.0)
            
            # Asymmetric budget split: give more to the function part (75%)
            # This encourages deeper nested lambdas and redexes
            split = int(budget * 0.75)
            
            if rng.random() < redex_prob and budget > 4:
                v_name = get_unique_name("r", len(ctx))
                # Explicit Redex: (Lam v. body) arg
                f_tm = ("Lam", v_name, arg_ty, gen_term(target_ty, ctx + [(v_name, arg_ty)], split))
                a_tm = gen_term(arg_ty, ctx, budget - split - 1)
                return ("App", f_tm, a_tm)
            else:
                # Standard application
                return ("App", 
                        gen_term(("Fun", arg_ty, target_ty), ctx, split), 
                        gen_term(arg_ty, ctx, budget - split - 1))

        else: # lit
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-999, 999))
            return ("String", f"s_{rng.randint(0, 999)}")

    # Generate sample with maximum allowed complexity
    # Target Ty size ~10, Term size ~100
    final_ty = gen_type(10)
    final_tm = gen_term(final_ty, [], 100)

    return final_ty, final_tm

import lc


def run() -> lc.RunOutput:
    """Run the analysis"""

    size = 1000
    rng = random.Random()

    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
    )
