<DESCRIPTION>
This generator combines Context-Aware Type Selection (CATS) with a Budget-Scaled Redex Probability and Cubic Recency Bias for variable selection. 

Key improvements:
1. **CATS**: 85% probability to select argument types from the existing context, ensuring that `App` nodes can more frequently resolve to `Var` or `Redex` nodes.
2. **Cubic Recency Bias**: Upgraded from quadratic to cubic bias `(index + 1)**3` to more aggressively favor local variables, which is crucial for maintaining high `used_vars` metrics in deep ASTs.
3. **Asymmetric 70/30 Budget**: Consistently applies the 70% budget rule to the function/body branch of applications and redexes, driving term depth.
4. **Dynamic Redex Injection**: The probability of injecting a redex scales with the remaining budget (`0.55 + budget/150`), ensuring complex structures early on.
5. **Forced Variable Arguments**: When a redex is created, there's a 90% probability to use a variable for the argument if a type-match exists in the context, directly boosting variable usage metrics.
</DESCRIPTION>

from typing import Literal, Tuple, List
import random
import tqdm

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]


type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    STLC generator optimized for high variable usage and redex density.
    Uses Cubic Recency Bias, Budget-Scaled Redex Injection, and CATS.
    """
    var_counter = 0

    def get_unique_name(prefix: str, depth: int) -> str:
        nonlocal var_counter
        var_counter += 1
        return f"{prefix}_{depth}_{var_counter}_{rng.randint(0, 100)}"

    def gen_type(size_limit: int) -> Ty:
        # Favor function types (85% bias) for structural complexity
        if size_limit <= 1 or rng.random() < 0.15:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        left_size = rng.randint(1, max(1, size_limit - 1))
        return ("Fun", gen_type(left_size), gen_type(size_limit - left_size - 1))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        # Filter variables by type
        valid_vars = [name for name, t in ctx if t == target_ty]

        # Base Case: Budget exhausted
        if budget <= 1:
            if valid_vars and rng.random() < 0.95:
                # Cubic Recency Bias: Strong preference for locally bound variables
                weights = [(i + 1)**3 for i in range(len(valid_vars))]
                return ("Var", rng.choices(valid_vars, weights=weights, k=1)[0])

            # Must return a valid term for the type
            if target_ty[0] == "Fun":
                v = get_unique_name("v", len(ctx))
                return ("Lam", v, target_ty[1], gen_term(target_ty[2], ctx + [(v, target_ty[1])], 0))

            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-99, 99))
            return ("String", "const")

        # Weighted Constructor Selection
        choices, weights = [], []
        if target_ty[0] == "Fun":
            choices.append("lam"); weights.append(30)
        if valid_vars:
            choices.append("var"); weights.append(40)
        if budget > 2:
            choices.append("app"); weights.append(60)
        
        if not choices or (target_ty[0] != "Fun" and rng.random() < 0.1):
            choices.append("lit"); weights.append(10)

        mode = rng.choices(choices, weights=weights)[0]

        if mode == "lam":
            v_name = get_unique_name("x", len(ctx))
            return ("Lam", v_name, target_ty[1], gen_term(target_ty[2], ctx + [(v_name, target_ty[1])], budget - 1))

        elif mode == "var":
            v_weights = [(i + 1)**3 for i in range(len(valid_vars))]
            return ("Var", rng.choices(valid_vars, weights=v_weights, k=1)[0])

        elif mode == "app":
            # 1. Context-Aware Type Selection (CATS)
            if ctx and rng.random() < 0.85:
                arg_ty = rng.choice(ctx)[1]
            else:
                arg_ty = gen_type(rng.randint(1, 2))

            # 2. Budget-Scaled Probability for Redexes
            redex_prob = 0.55 + (budget / 150.0)
            
            # 3. Asymmetric 70/30 Budget Split
            split = int(budget * 0.70)

            if rng.random() < redex_prob and budget > 4:
                v_name = get_unique_name("r", len(ctx))
                # Lambda side of redex
                f_tm = ("Lam", v_name, arg_ty, gen_term(target_ty, ctx + [(v_name, arg_ty)], split))
                
                # 4. Forced Variable Argument
                arg_vars = [n for n, t in ctx if t == arg_ty]
                if arg_vars and rng.random() < 0.90:
                    a_weights = [(i + 1)**3 for i in range(len(arg_vars))]
                    a_tm = ("Var", rng.choices(arg_vars, weights=a_weights, k=1)[0])
                else:
                    a_tm = gen_term(arg_ty, ctx, budget - split - 1)
                return ("App", f_tm, a_tm)
            else:
                # Standard application
                return ("App",
                        gen_term(("Fun", arg_ty, target_ty), ctx, split),
                        gen_term(arg_ty, ctx, budget - split - 1))

        else: # lit
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-1000, 1000))
            return ("String", f"s_{rng.randint(0, 999)}")

    # Entry point
    final_ty = gen_type(10)
    final_tm = gen_term(final_ty, [], 100)
    return final_ty, final_tm
