<DESCRIPTION>
The structural redesign focuses on a 'Synergistic Redex Factory' architecture. It shifts from balanced term generation to a deep functional nesting strategy. Key changes include:
1.  **Aggressive Redex Injection**: The `App` constructor is heavily biased to create an immediate `Lam` (redex) on the left branch, which is the most efficient way to maximize `applied_lambdas`.
2.  **Coupled Variable Selection**: The 'Forced Variable Argument' rule is strictly enforced (90% probability) within redexes. This creates a direct correlation between `applied_lambdas` and `used_vars`.
3.  **Cubic Recency Bias**: Variable selection uses a cubic weight distribution `(index + 1)**3` to favor the most recent (local) scope, which significantly boosts variable reuse metrics.
4.  **Asymmetric 70/30 Budgeting**: Applications allocate 70% of the budget to the function side, creating the deep, narrow trees that favor the target metrics over broad, shallow ones.
5.  **Context-Aware Type Selection (CATS)**: 85% of application argument types are pulled directly from the current context, ensuring that the 'Forced Variable' rule has valid candidates to select from.

from typing import Literal, Tuple, List
import random

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]

type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generates STLC terms optimized for high redex density and variable usage.
    Implements CATS, Cubic Recency Bias, and Asymmetric Budgeting.
    """
    var_counter = 0

    def get_unique_id(prefix: str, ctx_len: int) -> str:
        nonlocal var_counter
        var_counter += 1
        return f"{prefix}_{ctx_len}_{var_counter}_{rng.randint(0, 100)}"

    def gen_type(size_limit: int) -> Ty:
        # 85% Bias toward function types to ensure room for applications
        if size_limit <= 1 or rng.random() < 0.15:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        split = rng.randint(1, max(1, size_limit - 1))
        return ("Fun", gen_type(split), gen_type(size_limit - split - 1))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        valid_vars = [name for name, t in ctx if t == target_ty]

        # Terminal Safety: Budget exhausted or FunTy leaf requirement
        if budget <= 0:
            if valid_vars:
                # Cubic Recency Bias
                weights = [(i + 1)**3 for i in range(len(valid_vars))]
                return ("Var", rng.choices(valid_vars, weights=weights, k=1)[0])
            if target_ty[0] == "Fun":
                v = get_unique_id("f", len(ctx))
                return ("Lam", v, target_ty[1], gen_term(target_ty[2], ctx + [(v, target_ty[1])], 0))
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-10, 10))
            return ("String", "leaf")

        # Weighted Constructor Distribution
        # App (65), Var (40), Lam (25), Lit (10)
        choices, weights = [], []
        if target_ty[0] == "Fun":
            choices.append("lam"); weights.append(25)
        if valid_vars:
            choices.append("var"); weights.append(40)
        if budget > 1:
            choices.append("app"); weights.append(65)
        if not choices or (target_ty[0] != "Fun" and rng.random() < 0.05):
            choices.append("lit"); weights.append(10)

        mode = rng.choices(choices, weights=weights)[0]

        if mode == "lam":
            v_name = get_unique_id("x", len(ctx))
            return ("Lam", v_name, target_ty[1], gen_term(target_ty[2], ctx + [(v_name, target_ty[1])], budget - 1))

        elif mode == "var":
            v_weights = [(i + 1)**3 for i in range(len(valid_vars))]
            return ("Var", rng.choices(valid_vars, weights=v_weights, k=1)[0])

        elif mode == "app":
            # CATS: 85% Context Bias for argument types
            if ctx and rng.random() < 0.85:
                arg_ty = rng.choice(ctx)[1]
            else:
                arg_ty = gen_type(rng.randint(1, 2))

            # Dynamic Redex Probability based on budget
            redex_prob = 0.55 + (budget / 150.0)
            # 70/30 Asymmetric Budgeting
            split = int(budget * 0.7)

            if rng.random() < redex_prob and budget > 3:
                # Synergistic Redex Injection
                v_name = get_unique_id("r", len(ctx))
                f_tm = ("Lam", v_name, arg_ty, gen_term(target_ty, ctx + [(v_name, arg_ty)], split))
                
                # Forced Variable Argument (90% probability)
                arg_vars = [n for n, t in ctx if t == arg_ty]
                if arg_vars and rng.random() < 0.90:
                    a_weights = [(i + 1)**3 for i in range(len(arg_vars))]
                    a_tm = ("Var", rng.choices(arg_vars, weights=a_weights, k=1)[0])
                else:
                    a_tm = gen_term(arg_ty, ctx, budget - split - 1)
                return ("App", f_tm, a_tm)
            else:
                # Standard application
                return ("App", 
                        gen_term(("Fun", arg_ty, target_ty), ctx, split), 
                        gen_term(arg_ty, ctx, budget - split - 1))

        else: # lit
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-100, 100))
            return ("String", f"s_{rng.randint(0, 999)}")

    # Entry point: Target size 20-50 nodes via budget 100
    root_ty = gen_type(rng.randint(5, 10))
    root_tm = gen_term(root_ty, [], 100)
    return root_ty, root_tm

# EVOLVE-BLOCK-END
