<DESCRIPTION>
This generator maximizes performance metrics by prioritizing variable reuse and redex injection through frequency-based context analysis. 

Key improvements:
1. **Context-Driven Type Matching**: During `App` construction, instead of a random type, it analyzes the current context to select an `arg_ty` that appears most frequently. This significantly boosts `used_vars_proportions` by ensuring the argument side of an application can almost always be satisfied by a `Var`.
2. **Probability-Scaled Redex Injection**: Uses a dynamic redex injection probability calibrated to the remaining budget ($P = 0.6 + budget / 200$), ensuring a high density of applied lambdas in large terms.
3. **Quadratic Recency Bias**: Implements a power-law weighting $(index + 1)^2$ when selecting variables, favoring local scope and mimicking realistic code patterns.
4. **Enhanced Type Complexity**: Type generation is biased toward `FunTy` (85%) to ensure the term generation has enough structure to perform applications and abstractions.
5. **Strict Budget Management**: Uses an asymmetric budget split (75/25) to ensure deep, complex functional branches while maintaining valid, simple leaves.

from typing import Literal, Tuple, List
import random

"""
`Ty` and `Tm` define the shape of a simply-typed lambda-calculus deeply embedded in Python.
"""

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]


type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generates high-complexity STLC terms by matching application types to context 
    frequency and using scaled redex injection.
    """
    var_counter = 0

    def get_unique_name(depth: int) -> str:
        nonlocal var_counter
        var_counter += 1
        return f"v{depth}_{var_counter}_{rng.randint(0, 999)}"

    def gen_type(size_limit: int) -> Ty:
        # Bias heavily towards functions to facilitate more complex structures
        if size_limit <= 1 or rng.random() < 0.15:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        split = rng.randint(1, max(1, size_limit - 1))
        return ("Fun", gen_type(split), gen_type(size_limit - split - 1))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        # Filter variables by type
        valid_vars = [name for name, t in ctx if t == target_ty]

        # Terminal logic
        if budget <= 1:
            if valid_vars and rng.random() < 0.95:
                # Recency bias: favor variables near the end of the context (most local)
                weights = [(i + 1)**2 for i in range(len(valid_vars))]
                return ("Var", rng.choices(valid_vars, weights=weights, k=1)[0])
            
            # Validity fallback for function types
            if target_ty[0] == "Fun":
                v = get_unique_name(len(ctx))
                return ("Lam", v, target_ty[1], gen_term(target_ty[2], ctx + [(v, target_ty[1])], 0))
            
            # Literal fallbacks
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-50, 50))
            return ("String", rng.choice(["a", "b", "c"]))

        # Biased constructor weights
        choices, weights = [], []
        if target_ty[0] == "Fun":
            choices.append("lam"); weights.append(30)
        if valid_vars:
            choices.append("var"); weights.append(35)
        if budget > 2:
            choices.append("app"); weights.append(65)
        
        if not choices or rng.random() < 0.1:
            choices.append("lit"); weights.append(10)

        mode = rng.choices(choices, weights=weights)[0]

        if mode == "lam":
            v_name = get_unique_name(len(ctx))
            return ("Lam", v_name, target_ty[1], gen_term(target_ty[2], ctx + [(v_name, target_ty[1])], budget - 1))

        elif mode == "var":
            v_weights = [(i + 1)**2 for i in range(len(valid_vars))]
            return ("Var", rng.choices(valid_vars, weights=v_weights, k=1)[0])

        elif mode == "app":
            # IMPROVED: Type-Matching for Variables. 
            # Look for types that already exist in the context to increase used_vars metric.
            if ctx and rng.random() < 0.8:
                arg_ty = rng.choice(ctx)[1]
            else:
                arg_ty = gen_type(rng.randint(1, 2))

            # Redex Injection (Explicitly create applied lambdas)
            # Probability increases as budget increases
            redex_p = 0.6 + (budget / 200.0)
            split = int(budget * 0.75)  # Asymmetric split favored for deep Lam bodies

            if rng.random() < redex_p and budget > 4:
                v_name = get_unique_name(len(ctx))
                f_tm = ("Lam", v_name, arg_ty, gen_term(target_ty, ctx + [(v_name, arg_ty)], split))
                
                # Check if we can satisfy the argument branch with a Var immediately
                arg_vars = [n for n, t in ctx if t == arg_ty]
                if arg_vars and rng.random() < 0.9:
                    a_weights = [(i + 1)**2 for i in range(len(arg_vars))]
                    a_tm = ("Var", rng.choices(arg_vars, weights=a_weights, k=1)[0])
                else:
                    a_tm = gen_term(arg_ty, ctx, budget - split - 1)
                return ("App", f_tm, a_tm)
            else:
                # Standard application path
                return ("App", 
                        gen_term(("Fun", arg_ty, target_ty), ctx, split), 
                        gen_term(arg_ty, ctx, budget - split - 1))

        else: # literal
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-1000, 1000))
            return ("String", f"s{rng.randint(0, 5000)}")

    # Objective: Maximize type structure and term depth
    root_ty = gen_type(10)
    root_tm = gen_term(root_ty, [], 100)

    return root_ty, root_tm

# EVOLVE-BLOCK-END
