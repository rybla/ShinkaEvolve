<DESCRIPTION>
This algorithm optimizes STLC term generation by focusing on three key metrics: variable usage, applied lambdas, and structural complexity. 

1.  **Quadratic Recency Bias**: Instead of simple power-law indexing, it uses a quadratic weight `(i+1)**2` for variable selection from the context, significantly boosting `used_vars_proportions`.
2.  **Aggressive Redex Injection**: The probability of generating an applied lambda (`App(Lam, Arg)`) scales dynamically with the remaining budget, ensuring that larger terms are saturated with redexes.
3.  **Context-Driven Type Selection**: When generating an application, the argument type is selected from the existing context with 75% probability. This creates a "closed-loop" where the argument branch is highly likely to be satisfied by a variable already in scope.
4.  **Asymmetric Budgeting**: Application nodes split the budget 70/30 to favor deeper, more complex function branches, which increases the average term size and unique sub-term count.
</DESCRIPTION>

from typing import Literal, Tuple, List
import random

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]

type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generates well-typed STLC terms optimized for high variable usage, 
    frequent applied lambdas, and structural uniqueness.
    """
    
    var_id = 0

    def get_name(prefix: str) -> str:
        nonlocal var_id
        var_id += 1
        return f"{prefix}_{var_id}_{rng.randint(0, 999)}"

    def gen_type(size_limit: int) -> Ty:
        # Bias towards function types to allow for more complex term structures
        if size_limit <= 1 or rng.random() < 0.15:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        split = rng.randint(1, max(1, size_limit - 1))
        return ("Fun", gen_type(split), gen_type(size_limit - split - 1))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        # Identify variables in context that match the required type
        valid_vars = [name for name, t in ctx if t == target_ty]

        # Base Case: Budget exhausted or forced leaf
        if budget <= 1:
            if valid_vars and rng.random() < 0.95:
                # Quadratic Recency Bias: Favor variables added most recently to the context
                # This significantly boosts used_vars_proportions_average
                weights = [(i + 1)**2 for i in range(len(valid_vars))]
                return ("Var", rng.choices(valid_vars, weights=weights, k=1)[0])
            
            if target_ty[0] == "Fun":
                v = get_name("v")
                return ("Lam", v, target_ty[1], gen_term(target_ty[2], ctx + [(v, target_ty[1])], 0))
            
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-100, 100))
            return ("String", rng.choice(["a", "b", "c"]))

        # Weighted Constructor Selection
        choices, weights = [], []
        
        if target_ty[0] == "Fun":
            choices.append("lam")
            weights.append(25)
        
        if valid_vars:
            choices.append("var")
            weights.append(40) # High weight to ensure variables are used
            
        if budget > 2:
            choices.append("app")
            weights.append(65) # Highest weight to drive size and redexes
            
        if not choices or (target_ty[0] != "Fun" and rng.random() < 0.1):
            choices.append("lit")
            weights.append(10)

        mode = rng.choices(choices, weights=weights)[0]

        if mode == "lam":
            v_name = get_name("x")
            return ("Lam", v_name, target_ty[1], gen_term(target_ty[2], ctx + [(v_name, target_ty[1])], budget - 1))

        elif mode == "var":
            v_weights = [(i + 1)**2 for i in range(len(valid_vars))]
            return ("Var", rng.choices(valid_vars, weights=v_weights, k=1)[0])

        elif mode == "app":
            # Context-Aware Argument Selection (75% chance)
            # Picking a type already in ctx makes it likely the arg branch can use a Var
            if ctx and rng.random() < 0.75:
                arg_ty = rng.choice(ctx)[1]
            else:
                arg_ty = gen_type(rng.randint(1, 3))

            # Scaled Redex Injection: Probability increases with budget
            # This maximizes applied_lambdas_proportions_average
            redex_prob = 0.45 + (budget / 180.0)
            
            # Asymmetric budget split to encourage deeper trees
            split = int(budget * 0.7)
            
            if rng.random() < redex_prob and budget > 4:
                v_name = get_name("r")
                # Construct (Lam x. body) arg
                f_tm = ("Lam", v_name, arg_ty, gen_term(target_ty, ctx + [(v_name, arg_ty)], split))
                a_tm = gen_term(arg_ty, ctx, budget - split - 1)
                return ("App", f_tm, a_tm)
            else:
                # Standard application
                return ("App", 
                        gen_term(("Fun", arg_ty, target_ty), ctx, split), 
                        gen_term(arg_ty, ctx, budget - split - 1))

        else: # lit
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-1000, 1000))
            return ("String", f"s_{rng.randint(0, 9999)}")

    # Generate sample with maximum allowed complexity
    # Target type size ~10, Term budget ~100
    final_ty = gen_type(10)
    final_tm = gen_term(final_ty, [], 100)

    return final_ty, final_tm
