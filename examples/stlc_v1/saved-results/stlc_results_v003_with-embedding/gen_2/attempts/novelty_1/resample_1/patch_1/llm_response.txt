<DESCRIPTION>
The previous generator was extremely limited, producing mostly constants and simple identity-like functions with zero variable usage or applications. 

This update introduces a robust, goal-oriented generator that:
1. Generates complex types (up to 10 nodes) using a budget-based approach.
2. Generates well-typed terms of a target type within a specific context (variable environment).
3. Prefers `App` and `Lam` nodes to increase term size and complexity.
4. Intelligently selects variables from the context to maximize variable usage.
5. Implements a size-tracking mechanism to stay within the 100-node limit while pushing for larger terms.
</DESCRIPTION>

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generate a well-typed sample type and term using a random seed.
    """

    def get_ty_size(ty: Ty) -> int:
        if ty[0] == "Fun":
            return 1 + get_ty_size(ty[1]) + get_ty_size(ty[2])
        return 1

    def gen_type(budget: int) -> Ty:
        if budget <= 1:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        if rng.random() < 0.7:
            left_budget = rng.randint(1, budget - 2)
            return ("Fun", gen_type(left_budget), gen_type(budget - 1 - left_budget))
        else:
            return rng.choice([("Bool",), ("Int",), ("String",)])

    def gen_term(target_ty: Ty, ctx: list[Tuple[str, Ty]], budget: int) -> Tm:
        # Try to use a variable from context if it matches the type
        matching_vars = [name for name, ty in ctx if ty == target_ty]
        
        # Base case or random leaf
        if budget <= 1:
            if matching_vars and (rng.random() < 0.7 or target_ty[0] == "Fun"):
                return ("Var", rng.choice(matching_vars))
            if target_ty == ("Bool",):
                return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",):
                return ("Int", rng.randint(0, 100))
            if target_ty == ("String",):
                return ("String", "s")
            # If function type and no var, must lambda (unlikely with budget 1 but defensive)
            return gen_lam(target_ty, ctx, budget)

        # Probabilistic choice of term constructor
        choices = []
        if target_ty[0] == "Fun":
            choices.append("Lam")
        if budget > 3:
            choices.append("App")
        if matching_vars:
            choices.append("Var")
        
        # Weighted choice: prefer App and Lam to increase size and complexity
        weights = {"Lam": 0.4, "App": 0.5, "Var": 0.1}
        valid_choices = [c for c in choices if c in weights]
        if not valid_choices:
            return gen_term(target_ty, ctx, 0) # fallback to leaf
            
        choice = rng.choices(valid_choices, weights=[weights[c] for c in valid_choices])[0]

        if choice == "Lam":
            return gen_lam(target_ty, ctx, budget)
        elif choice == "App":
            # Generate an application: f x where f: A -> target_ty and x: A
            arg_ty = gen_type(rng.randint(1, 4))
            fun_ty = ("Fun", arg_ty, target_ty)
            f_tm = gen_term(fun_ty, ctx, budget // 2)
            x_tm = gen_term(arg_ty, ctx, budget // 2)
            return ("App", f_tm, x_tm)
        else:
            return ("Var", rng.choice(matching_vars))

    def gen_lam(target_ty: Ty, ctx: list[Tuple[str, Ty]], budget: int) -> Tm:
        if target_ty[0] != "Fun":
            # Coerce: if we need a base type but want to lam, generate ((\x.x) val)
            inner_ty = ("Bool",)
            var_name = f"v{len(ctx)}"
            id_lam = ("Lam", var_name, inner_ty, ("Var", var_name))
            val = gen_term(inner_ty, ctx, 1)
            # This is complex, simplify: just give a constant
            return gen_term(target_ty, ctx, 0)
        
        var_name = f"v{len(ctx)}"
        param_ty = target_ty[1]
        ret_ty = target_ty[2]
        body = gen_term(ret_ty, ctx + [(var_name, param_ty)], budget - 1)
        return ("Lam", var_name, param_ty, body)

    # Goal: Type size ~7-10, Term size ~50-100
    chosen_ty = gen_type(rng.randint(5, 10))
    chosen_tm = gen_term(chosen_ty, [], 80)
    
    return chosen_ty, chosen_tm
