<DESCRIPTION>
The generator implements a high-performance STLC term synthesizer focusing on the synergy between redexes and variable reuse. 
Key architectural choices:
1. **Cubic Recency Bias**: Variable selection uses `(index + 1)**3` weighting to aggressively favor local scope, which is statistically more likely to be used in human-like or complex generated terms, boosting the `used_vars` metric.
2. **Context-Aware Type Selection (CATS)**: 85% of `App` nodes select argument types from types already present in the environment.
3. **Budget-Dependent Constructor Weighting**: The probability of `App` and `Lam` nodes scales with the remaining budget to prevent early termination and ensure deep nesting.
4. **Strong Redex Injection**: Redexes are treated as the primary mechanism for combining metrics, with a 90% "Forced Variable Argument" rule when a redex is created, ensuring that most applied lambdas also contribute to variable usage scores.
5. **Asymmetric 70/30 Budgeting**: Prevents shallow trees by allocating the majority of the budget to function bodies.

from typing import Literal, Tuple, List
import random

"""
`Ty` and `Tm` define the shape of a simply-typed lambda-calculus deeply embedded in Python.
"""

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]

type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generates well-typed STLC terms using Cubic Recency Bias and Redex-Variable Synergy.
    """
    var_counter = 0

    def get_unique_name(prefix: str, depth: int) -> str:
        nonlocal var_counter
        var_counter += 1
        return f"{prefix}_{depth}_{var_counter}_{rng.randint(0, 100)}"

    def gen_type(size_limit: int) -> Ty:
        # Bias towards functions to allow complex logic (size 5-10 target for root)
        if size_limit <= 1 or rng.random() < 0.15:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        
        split = rng.randint(1, max(1, size_limit - 1))
        return ("Fun", gen_type(split), gen_type(size_limit - split - 1))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        valid_vars = [name for name, t in ctx if t == target_ty]

        # Base case: Force variable or literal if budget depleted
        if budget <= 1:
            if valid_vars and rng.random() < 0.95:
                # Cubic Recency Bias: Stronger preference for recent scope
                weights = [(i + 1)**3 for i in range(len(valid_vars))]
                return ("Var", rng.choices(valid_vars, weights=weights, k=1)[0])
            
            # Type-safe fallback construction
            if target_ty[0] == "Fun":
                v = get_unique_name("y", len(ctx))
                return ("Lam", v, target_ty[1], gen_term(target_ty[2], ctx + [(v, target_ty[1])], 0))
            
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-10, 10))
            return ("String", "leaf")

        # Dynamic Constructor Weights: Focus on App (60) and Var (40)
        choices, weights = [], []
        if target_ty[0] == "Fun":
            choices.append("lam"); weights.append(25)
        if valid_vars:
            choices.append("var"); weights.append(40)
        if budget > 2:
            choices.append("app"); weights.append(60)
        
        if not choices or (target_ty[0] != "Fun" and rng.random() < 0.1):
            choices.append("lit"); weights.append(10)

        mode = rng.choices(choices, weights=weights)[0]

        if mode == "lam":
            v_name = get_unique_name("x", len(ctx))
            return ("Lam", v_name, target_ty[1], gen_term(target_ty[2], ctx + [(v_name, target_ty[1])], budget - 1))

        elif mode == "var":
            v_weights = [(i + 1)**3 for i in range(len(valid_vars))]
            return ("Var", rng.choices(valid_vars, weights=v_weights, k=1)[0])

        elif mode == "app":
            # Primary Driver: Context-Aware Type Selection (CATS)
            if ctx and rng.random() < 0.85:
                arg_ty = rng.choice(ctx)[1]
            else:
                arg_ty = gen_type(rng.randint(1, 2))

            # Metric synergy: Probability of generating a redex scales with budget
            red_prob = 0.55 + (budget / 150.0)
            split = int(budget * 0.7) # Asymmetric 70/30 Rule

            if rng.random() < red_prob and budget > 4:
                v_name = get_unique_name("r", len(ctx))
                # Body of the lambda gets the bulk of the budget
                f_tm = ("Lam", v_name, arg_ty, gen_term(target_ty, ctx + [(v_name, arg_ty)], split))
                
                # Forced Variable Argument: 90% chance to reuse variable
                arg_vars = [n for n, t in ctx if t == arg_ty]
                if arg_vars and rng.random() < 0.90:
                    a_weights = [(i + 1)**3 for i in range(len(arg_vars))]
                    a_tm = ("Var", rng.choices(arg_vars, weights=a_weights, k=1)[0])
                else:
                    a_tm = gen_term(arg_ty, ctx, budget - split - 1)
                return ("App", f_tm, a_tm)
            else:
                # Standard application
                return ("App", 
                        gen_term(("Fun", arg_ty, target_ty), ctx, split), 
                        gen_term(arg_ty, ctx, budget - split - 1))

        else: # lit
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-999, 999))
            return ("String", f"s_{rng.randint(0, 999)}")

    # Initialize with significant complexity
    root_ty = gen_type(10)
    root_tm = gen_term(root_ty, [], 100)
    return root_ty, root_tm

# EVOLVE-BLOCK-END

import lc

def run() -> lc.RunOutput:
    """Run the analysis"""
    size = 1000
    rng = random.Random()
    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
    )
