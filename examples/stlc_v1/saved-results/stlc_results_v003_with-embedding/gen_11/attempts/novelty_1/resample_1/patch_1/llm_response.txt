<DESCRIPTION>
This algorithm improves upon previous versions by implementing three key strategies:
1. **Global Uniqueness & Shadowing Prevention**: Uses a global counter for variable names. This ensures that every variable in the context is uniquely reachable, preventing the "shadowing" that reduces variable usage scores in standard recursive generators.
2. **Type-Informed Context Sampling**: When generating an `App` node, the generator doesn't just create a random argument type. It specifically looks at the types already available in the `ctx`. By choosing an `arg_ty` that matches existing variables, it significantly increases the probability that the argument branch can be a `Var` rather than a `Lit`, boosting the `used_vars` metric.
3. **Aggressive Redex Injection with Budget Scaling**: Instead of a flat probability, redex injection (creating a `Lam` and immediately applying it) is prioritized when the budget is high. This ensures the `applied_lambdas` metric is maximized while still allowing the tree to grow deep.
4. **Recency-Biased Variable Selection**: Uses a power-law distribution to pick variables from the context, favoring the most recently bound ones to mimic realistic code patterns.
</DESCRIPTION>

from typing import Literal, Tuple, List, Optional
import random

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]

type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generates a well-typed STLC term using a context-aware strategy 
    to maximize variable usage, redexes, and sub-term uniqueness.
    """
    
    # Global counter to prevent variable shadowing and maximize unique subterms
    var_counter = 0

    def gen_type(size_limit: int) -> Ty:
        if size_limit <= 1 or rng.random() < 0.2:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        # Favor function types to allow for more complex terms
        split = rng.randint(1, max(1, size_limit - 1))
        return ("Fun", gen_type(split), gen_type(size_limit - split - 1))

    def gen_term(target_ty: Ty, ctx: List[Tuple[str, Ty]], budget: int) -> Tm:
        nonlocal var_counter
        
        # Filter context for type-safe variables
        valid_vars = [name for name, t in ctx if t == target_ty]

        # Base case for low budget
        if budget <= 1:
            if valid_vars:
                # Recency bias: prefer variables defined later (higher index)
                idx = int(len(valid_vars) * (rng.random() ** 0.3))
                return ("Var", valid_vars[::-1][idx])
            
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-999, 999))
            if target_ty == ("String",): return ("String", rng.choice(["alpha", "beta", "gamma"]))
            
            if target_ty[0] == "Fun":
                var_counter += 1
                v = f"v_{var_counter}"
                return ("Lam", v, target_ty[1], gen_term(target_ty[2], ctx + [(v, target_ty[1])], 0))
            return ("Int", 0)

        # Weighted selection of term constructors
        choices, weights = [], []
        
        if target_ty[0] == "Fun":
            choices.append("lam"); weights.append(40)
        
        if valid_vars:
            choices.append("var"); weights.append(35)
            
        if budget > 2:
            choices.append("app"); weights.append(50)
            
        if target_ty[0] != "Fun" or not choices:
            choices.append("lit"); weights.append(10)

        mode = rng.choices(choices, weights=weights)[0]

        if mode == "lam":
            var_counter += 1
            v_name = f"x_{var_counter}"
            return ("Lam", v_name, target_ty[1], gen_term(target_ty[2], ctx + [(v_name, target_ty[1])], budget - 1))

        elif mode == "var":
            idx = int(len(valid_vars) * (rng.random() ** 0.3))
            return ("Var", valid_vars[::-1][idx])

        elif mode == "app":
            # Strategy: Pick an arg_ty that we already have in context to increase Var usage
            if ctx and rng.random() < 0.6:
                arg_ty = rng.choice(ctx)[1]
            else:
                arg_ty = gen_type(rng.randint(1, 3))

            # Split budget: give slightly more to the function side
            split = rng.randint(1, budget - 1)
            
            # Explicit Redex Injection (App of a Lam)
            # High probability (50%) to ensure applied_lambdas_proportions is high
            if rng.random() < 0.5 and budget > 5:
                var_counter += 1
                v_name = f"r_{var_counter}"
                f_tm = ("Lam", v_name, arg_ty, gen_term(target_ty, ctx + [(v_name, arg_ty)], split))
                a_tm = gen_term(arg_ty, ctx, budget - split - 1)
                return ("App", f_tm, a_tm)
            else:
                return ("App", 
                        gen_term(("Fun", arg_ty, target_ty), ctx, split), 
                        gen_term(arg_ty, ctx, budget - split - 1))

        else: # lit
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(-1000, 1000))
            return ("String", f"s_{rng.randint(0, 9999)}")

    # Generate a complex type (size ~10) and a large term (budget 100)
    final_ty = gen_type(10)
    final_tm = gen_term(final_ty, [], 100)

    return final_ty, final_tm

# EVOLVE-BLOCK-END
