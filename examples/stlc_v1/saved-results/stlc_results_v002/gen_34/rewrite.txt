import random
from typing import Literal, Tuple

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]

type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

# EVOLVE-BLOCK-START

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generates STLC terms using tiered type-driven synthesis.
    Focuses on maximizing variable usage by matching argument types to current context.
    """

    def gen_type(nodes: int) -> Ty:
        if nodes <= 1:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        # High bias towards functions to ensure we keep pushing variables into the context
        if rng.random() < 0.75:
            return ("Fun", gen_type(1), gen_type(nodes - 1))
        return rng.choice([("Bool",), ("Int",), ("String",)])

    def gen_term(ctx: list[Tuple[str, Ty]], target_ty: Ty, budget: int) -> Tm:
        # 1. Tier 1: Variable Match (Prioritize used_vars)
        matching_vars = [n for n, t in ctx if t == target_ty]
        if matching_vars and (budget <= 2 or rng.random() < 0.45):
            return ("Var", rng.choice(matching_vars))

        # 2. Tier 2: Redex-Chain Injection (The Applied Lambda Multiplier)
        # We pick types already 'inhabited' in the context to ensure args are 'Var' nodes
        if budget > 25 and rng.random() < 0.4:
            num_args = rng.randint(2, 4 if budget > 60 else 3)
            arg_types = []
            for _ in range(num_args):
                if ctx and rng.random() < 0.9: # High inhabited bias
                    arg_types.append(rng.choice(ctx)[1])
                else:
                    arg_types.append(gen_type(1))
            
            # Sub-chain generation
            sub_ctx = list(ctx)
            vnames = []
            for i in range(num_args):
                vname = f"v{len(ctx) + i}"
                vnames.append(vname)
                sub_ctx.append((vname, arg_types[i]))
            
            # The innermost body
            res = gen_term(sub_ctx, target_ty, int(budget * 0.5))
            
            # Wrap in Lambdas
            for i in reversed(range(num_args)):
                res = ("Lam", vnames[i], arg_types[i], res)
            
            # Apply to arguments (kept small to ensure they stay as Vars/Lits)
            for i in range(num_args):
                res = ("App", res, gen_term(ctx, arg_types[i], 2))
            return res

        # 3. Tier 3: Callable Lookup with Recursive Discovery
        # If we have f: A -> B and we want B, try App(f, gen(A))
        if budget > 4 and ctx:
            callables = [(n, t[1]) for n, t in ctx if t[0] == "Fun" and t[2] == target_ty]
            if callables and rng.random() < 0.6:
                vname, arg_ty = rng.choice(callables)
                return ("App", ("Var", vname), gen_term(ctx, arg_ty, budget - 2))

        # 4. Tier 4: Mandatory Abstraction for Function Types
        if target_ty[0] == "Fun":
            _, arg_ty, ret_ty = target_ty
            vname = f"v{len(ctx)}"
            return ("Lam", vname, arg_ty, gen_term(ctx + [(vname, arg_ty)], ret_ty, budget - 1))

        # 5. Tier 5: Directed Application
        if budget > 3:
            # Prefer types we already have variables for
            if ctx and rng.random() < 0.8:
                arg_ty = rng.choice(ctx)[1]
            else:
                arg_ty = gen_type(1)
            
            fun_ty = ("Fun", arg_ty, target_ty)
            # Skew budget heavily to the left to promote Lam/App inside the operator
            return ("App", 
                    gen_term(ctx, fun_ty, int(budget * 0.75)), 
                    gen_term(ctx, arg_ty, int(budget * 0.15)))

        # 6. Safety Leaves
        if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
        if target_ty == ("Int",): return ("Int", rng.randint(0, 42))
        if target_ty == ("String",): return ("String", "id")
        return ("Int", 1)

    # Entry Point
    target_type = gen_type(rng.randint(8, 11))
    # Optimized budget for average size 25-30
    term = gen_term([], target_type, 120)
    return target_type, term

# EVOLVE-BLOCK-END

import lc

def run() -> lc.RunOutput:
    """Run the analysis"""
    size = 1000
    rng = random.Random()
    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
    )
