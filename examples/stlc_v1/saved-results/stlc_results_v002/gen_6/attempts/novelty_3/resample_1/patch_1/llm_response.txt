@dataclass
class GenConfig:
    """Configuration for generation"""
    max_type_size: int = 10
    max_term_size: int = 100
    min_term_size: int = 5

class STLCGenerator:
    """Generates well-typed STLC terms with size control"""
    
    def __init__(self, rng: random.Random, config: GenConfig = None):
        self.rng = rng
        self.config = config or GenConfig()
        self.var_counter = 0
        self.used_vars: Dict[str, bool] = {}
        
    def fresh_var(self) -> str:
        """Generate a fresh variable name"""
        var = f"x{self.var_counter}"
        self.var_counter += 1
        self.used_vars[var] = False
        return var
    
    def generate_type(self, size_budget: int) -> Ty:
        """Generate a type with size control"""
        if size_budget <= 0:
            return self._generate_base_type()
        
        # Randomly choose type constructor
        choice = self.rng.randrange(0, 4)
        
        if choice == 0:
            return ("Bool",)
        elif choice == 1:
            return ("Int",)
        elif choice == 2:
            return ("String",)
        else:
            # Function type - consume more budget
            arg_size = self.rng.randrange(1, size_budget // 2 + 1)
            ret_size = size_budget - arg_size - 1
            return ("Fun", self.generate_type(arg_size), self.generate_type(ret_size))
    
    def _generate_base_type(self) -> Ty:
        """Generate a base type"""
        choice = self.rng.randrange(0, 3)
        if choice == 0:
            return ("Bool",)
        elif choice == 1:
            return ("Int",)
        else:
            return ("String",)
    
    def generate_term(self, expected_ty: Ty, size_budget: int, context: Dict[str, Ty]) -> Tm:
        """Generate a term with given expected type and size budget"""
        if size_budget <= 1:
            return self._generate_literal(expected_ty)
        
        # Decide between lambda, application, variable, or literal
        choice = self.rng.randrange(0, 4)
        
        if choice == 0 and isinstance(expected_ty, tuple) and expected_ty[0] == "Fun":
            return self._generate_lambda(expected_ty, size_budget, context)
        elif choice == 1 and context:
            return self._generate_application(expected_ty, size_budget, context)
        elif context and self.rng.random() < 0.3:
            return self._generate_var(expected_ty, context)
        else:
            return self._generate_literal(expected_ty)
    
    def _generate_lambda(self, fun_ty: Ty, size_budget: int, context: Dict[str, Ty]) -> Tm:
        """Generate a lambda term"""
        arg_var = self.fresh_var()
        arg_ty = fun_ty[1]
        new_context = context.copy()
        new_context[arg_var] = arg_ty
        
        # Generate body with remaining budget
        body_size = size_budget - 2  # Lambda node + var node
        body = self.generate_term(fun_ty[2], body_size, new_context)
        
        return ("Lam", arg_var, arg_ty, body)
    
    def _generate_application(self, app_ty: Ty, size_budget: int, context: Dict[str, Ty]) -> Tm:
        """Generate an application term"""
        # Find a function in context that can produce app_ty
        available_funcs = [(v, t) for v, t in context.items() if isinstance(t, tuple) and t[0] == "Fun"]
        
        if not available_funcs:
            return self._generate_literal(app_ty)
        
        # Pick random function
        func_var, func_ty = self.rng.choice(available_funcs)
        
        # Generate argument
        arg_ty = func_ty[1]
        arg_size = size_budget // 2
        arg = self.generate_term(arg_ty, arg_size, context)
        
        # Generate function (variable reference)
        func = ("Var", func_var)
        self.used_vars[func_var] = True
        
        return ("App", func, arg)
    
    def _generate_var(self, var_ty: Ty, context: Dict[str, Ty]) -> Tm:
        """Generate a variable reference"""
        # Find variables with matching type
        matching = [v for v, t in context.items() if self._types_equal(t, var_ty)]
        
        if not matching:
            return self._generate_literal(var_ty)
        
        var = self.rng.choice(matching)
        self.used_vars[var] = True
        return ("Var", var)
    
    def _generate_literal(self, ty: Ty) -> Tm:
        """Generate a literal based on type"""
        if ty == ("Bool",):
            return ("Bool", self.rng.choice([True, False]))
        elif ty == ("Int",):
            return ("Int", self.rng.randint(0, 100))
        elif ty == ("String",):
            return ("String", self.rng.choice(["hello", "world", "test"]))
        else:
            # Fallback for function types
            return ("Bool", True)
    
    def _types_equal(self, ty1: Ty, ty2: Ty) -> bool:
        """Check type equality"""
        if isinstance(ty1, tuple) and isinstance(ty2, tuple):
            if ty1[0] == "Fun" and ty2[0] == "Fun":
                return self._types_equal(ty1[1], ty2[1]) and self._types_equal(ty1[2], ty2[2])
            return ty1 == ty2
        return False
