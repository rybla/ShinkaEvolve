def generate_term(rng: random.Random, expected_ty: Ty, context: dict, term_depth: int) -> Tm:
    """Generate a term with the expected type, using variables from context."""
    # Handle base types
    if expected_ty[0] == "Bool" or expected_ty[0] == "Int" or expected_ty[0] == "String":
        if term_depth > 0 and context and rng.random() < 0.3:
            # Use a variable if available
            var_name = rng.choice(list(context.keys()))
            var_ty = context[var_name]
            if var_ty == expected_ty:
                return ("Var", var_name)
        # Otherwise create a literal
        return generate_literal(rng, expected_ty)
    
    # Handle function types
    if expected_ty[0] == "Fun":
        if term_depth > 0 and context and rng.random() < 0.3:
            # Apply a variable to an argument
            dom_ty = expected_ty[1]
            arg = generate_term(rng, dom_ty, context, term_depth - 1)

def generate_literal(rng: random.Random, ty: Ty) -> Tm:
