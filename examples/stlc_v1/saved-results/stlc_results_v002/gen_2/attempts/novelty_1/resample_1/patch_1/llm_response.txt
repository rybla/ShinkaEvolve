        # Base types

            return ("Bool",)

    else:
        # Function type

def generate_term(rng: random.Random, depth: int, ty: Ty, scope: List[Tuple[str, Ty]]) -> Tm:
    # Base case: if depth is exhausted or we hit a random chance, generate a literal for base types
    if depth <= 0 or (ty[0] == "Bool" and rng.random() < 0.5):
        return generate_literal(rng, ty)
    
    # For function types, decide whether to build a lambda or use something from scope
    if ty[0] == "Fun":
        # Higher chance to create a lambda when we have more depth available
        if depth > 1 and rng.random() < 0.7:
            # Generate parameter name and create lambda
            param_name = f"x{rng.randint(0, 100)}"
            param_ty = ty[1]
            body = generate_term(rng, depth - 1, ty[2], scope + [(param_name, param_ty)])
            return ("Lam", param_name, param_ty, body)
        # Otherwise try to use a variable from scope that has matching type
        else:
            # Try to find a variable in scope with compatible type
            for var_name, var_ty in reversed(scope):
                if types_compatible(var_ty, ty):
                    return ("Var", var_name)
    
    # Fallback: generate a literal
    return generate_literal(rng, ty)

def generate_literal(rng: random.Random, ty: Ty) -> Tm:
    if ty[0] == "Bool":
        return ("Bool", rng.choice([True, False]))
    elif ty[0] == "Int":
        return ("Int", rng.randint(0, 100))
    else:
        return ("String", "hello")

def types_compatible(ty1: Ty, ty2: Ty) -> bool:
