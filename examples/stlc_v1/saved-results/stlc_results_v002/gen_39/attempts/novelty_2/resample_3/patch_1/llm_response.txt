def generate_sample(rng: random.Random, depth=10, bound_vars: list[tuple[str, Ty]] = None) -> Tuple[Ty, Tm]:
    if bound_vars is None:
        bound_vars = []
    
    # Probabilistic base case termination
    if depth <= 0 or (not bound_vars and rng.random() < 0.4):
        i = rng.randrange(0, 3)
        if i == 0:
            return (("Bool",), ("Bool", True))
        elif i == 1:
            return (("Int",), ("Int", 27))
        else:
            return (("String",), ("String", "hello world"))
    
    # Choose what to generate based on available context
    i = rng.randrange(0, 6)
    
    if i == 0:
        # Generate a Bool literal
        return (("Bool",), ("Bool", rng.choice([True, False])))
    elif i == 1:
        # Generate an Int literal
        return (("Int",), ("Int", rng.randint(0, 100)))
    elif i == 2:
        # Generate a String literal
        return (("String",), ("String", rng.choice(["hello", "world", "test"])))
    elif i == 3 and bound_vars:
        # Use a variable from context
        var_name, var_ty = rng.choice(bound_vars)
        return (var_ty, ("Var", var_name))
    elif i == 4:
        # Create a lambda with a fresh variable
        arg_ty = ("Int",)
        var_name = f"x{rng.randint(0, 100)}"
        ret_ty, body = generate_sample(rng, depth - 1, bound_vars + [(var_name, arg_ty)])
        return (
            ("Fun", arg_ty, ret_ty),
            ("Lam", var_name, arg_ty, body)
        )
    else:
        # Application: apply a function to an argument
        # First generate a function type
        func_ty, func_tm = generate_sample(rng, depth - 1, bound_vars)
        if func_ty[0] == "Fun":
            # Generate argument with matching type
            arg_ty = func_ty[1]
            arg_tm = generate_sample(rng, depth - 1, bound_vars)
            return (
                func_ty[2],
                ("App", func_tm, arg_tm)
            )
        else:
            # Not a function, return as is
            return (func_ty, func_tm)
