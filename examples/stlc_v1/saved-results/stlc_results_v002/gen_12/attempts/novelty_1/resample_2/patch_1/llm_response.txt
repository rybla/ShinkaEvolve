def generate_sample(rng: random.Random, depth=10) -> Tuple[Ty, Tm]:
    # At depth 0, only generate literals
    if depth == 0:
        i = rng.randrange(0, 3)
        if i == 0:
            return (("Bool",), ("Bool", True))
        elif i == 1:
            return (("Int",), ("Int", 27))
        else:
            return (("String",), ("String", "hello world"))
    
    # At deeper depths, have more options
    i = rng.randrange(0, 6)  # Increased range for more options
    
    if i == 0:
        # Generate a Bool literal
        return (("Bool",), ("Bool", rng.choice([True, False])))
    elif i == 1:
        # Generate an Int literal
        return (("Int",), ("Int", rng.randint(0, 100)))
    elif i == 2:
        # Generate a String literal
        return (("String",), ("String", rng.choice(["hello", "world", "test"])))
    elif i == 3:
        # Generate a lambda that uses its bound variable
        var_type = ("Int",)
        return (
            ("Fun", var_type, var_type),
            ("Lam", "x", var_type, ("Var", "x"))
        )
    elif i == 4:
        # Generate an application
        func_type = ("Fun", ("Int",), ("Int",))
        func = ("Lam", "f", func_type, ("Var", "f"))
        arg = ("Int", 5)
        return (("Int",), ("App", func, arg))
    else:
        # Generate a more complex term
        return generate_sample(rng, depth - 1)
