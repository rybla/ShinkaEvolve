```python
def generate_sample(rng: random.Random, depth=10, context=None) -> Tuple[Ty, Tm]:
    """
    Generate a sample type and term using a random seed.
    Uses context to track bound variables for potential Var usage.
    """
    if context is None:
        context = []
    
    i = rng.randrange(0, 4 if depth > 0 else 3)
    
    if i == 0:
        return (("Bool",), ("Bool", True))
    elif i == 1:
        return (("Int",), ("Int", 27))
    elif i == 2:
        return (("String",), ("String", "hello world"))
    else:
        # Generate function type and term
        # Use context to potentially reference bound variables
        if context and rng.random() < 0.3:  # 30% chance to use a variable
            var_name = rng.choice(context)
            return (
                ("Fun", ("Int",), ("Int",)),
                ("Lam", "x", ("Int",), ("Var", var_name)),
            )
        else:
            # Add "x" to context when creating lambda
            new_context = context + ["x"]
            ty, tm = generate_sample(rng, depth - 1, new_context)
            return (
                ("Fun", ("Int",), ty),
                ("Lam", "x", ("Int",), tm),
            )
```

The key fix is passing `new_context` (which includes "x") to the recursive call instead of the original `context`. This ensures that when we recurse into the lambda body, the newly bound variable "x" is available to reference. This should increase the `used_vars_proportions_average` metric since variables will actually be used in the generated terms.
