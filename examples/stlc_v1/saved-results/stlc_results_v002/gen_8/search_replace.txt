    memo = {}

    def gen_term(ctx: list[Tuple[str, Ty]], target_ty: Ty, budget: int) -> Tm:
        # 1. Try to reuse a term from the memo cache for this type (20% chance)
        if target_ty in memo and rng.random() < 0.2 and budget > 5:
            return memo[target_ty]

        # 2. High priority: Use a variable that matches the type
        matching_vars = [name for name, ty in ctx if ty == target_ty]
        if matching_vars and (budget <= 1 or rng.random() < 0.4):
            return ("Var", rng.choice(matching_vars))

        # 3. Generate Lambda if target is a function (Mandatory for Fun types to ensure depth)
        if target_ty[0] == "Fun":
            _, arg_ty, ret_ty = target_ty
            var_name = f"v{len(ctx)}"
            res = ("Lam", var_name, arg_ty, gen_term(ctx + [(var_name, arg_ty)], ret_ty, budget - 1))
            memo[target_ty] = res
            return res

        # 4. Redex Boost: Explicitly create (Lam. body) arg
        if budget > 15 and rng.random() < 0.4:
            arg_ty = gen_type(2)
            v_name = f"v{len(ctx)}"
            lam = ("Lam", v_name, arg_ty, gen_term(ctx + [(v_name, arg_ty)], target_ty, budget // 2))
            arg = gen_term(ctx, arg_ty, budget // 2)
            return ("App", lam, arg)

        # 5. Application using existing callables
        if budget > 5:
            callables = [name for name, ty in ctx if ty[0] == "Fun" and ty[2] == target_ty]
            if callables and rng.random() < 0.6:
                vname = rng.choice(callables)
                vty = next(ty for n, ty in ctx if n == vname)
                return ("App", ("Var", vname), gen_term(ctx, vty[1], budget - 2))

        if budget <= 1:
            if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
            if target_ty == ("Int",): return ("Int", rng.randint(0, 100))
            if target_ty == ("String",): return ("String", "s")

        # 6. Standard Application
        if budget > 4:
            arg_ty = gen_type(rng.randint(1, 2))
            fun_ty = ("Fun", arg_ty, target_ty)
            res = ("App", gen_term(ctx, fun_ty, int(budget * 0.7)), gen_term(ctx, arg_ty, int(budget * 0.2)))
            if target_ty not in memo: memo[target_ty] = res
            return res
