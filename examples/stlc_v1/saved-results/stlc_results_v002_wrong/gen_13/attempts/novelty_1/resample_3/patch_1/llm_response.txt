```python
def generate_sample(rng: random.Random, depth=10) -> Tuple[Ty, Tm]:
    # Base case: generate literals when depth is low
    if depth <= 0:
        i = rng.randrange(0, 3)
        if i == 0:
            return (("Bool",), ("Bool", True))
        elif i == 1:
            return (("Int",), ("Int", 27))
        else:
            return (("String",), ("String", "hello world"))
    
    # At deeper levels, introduce variables and applications
    i = rng.randrange(0, 6)
    
    # Use a variable reference
    if i == 0:
        return (("Int",), ("Var", "x"))
    
    # Apply a function to an argument
    elif i == 1:
        func_ty, func_tm = generate_sample(rng, depth - 1)
        arg_ty, arg_tm = generate_sample(rng, depth - 1)
        return (("Fun", arg_ty, func_ty), ("App", func_tm, arg_tm))
    
    # Create a lambda that uses its variable
    elif i == 2:
        body_ty, body_tm = generate_sample(rng, depth - 1)
        return (("Fun", ("Int",), body_ty), ("Lam", "x", ("Int",), body_tm))
    
    # Base literals
    else:
        j = rng.randrange(0, 3)
        if j == 0:
            return (("Bool",), ("Bool", True))
        elif j == 1:
            return (("Int",), ("Int", 27))
        else:
            return (("String",), ("String", "hello world"))
```
