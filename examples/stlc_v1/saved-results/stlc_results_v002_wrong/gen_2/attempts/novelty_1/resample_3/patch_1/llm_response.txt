```python
def generate_sample(rng: random.Random, depth=10) -> Tuple[Ty, Tm]:
    """
    Generate a sample type and term using a random seed.
    """
    
    # First generate a type
    ty = generate_type(rng, depth)
    
    # Then generate a term with that type
    tm = generate_term(rng, depth, ty, [])
    
    return ty, tm

def generate_type(rng, depth):
    """Generate a random type up to certain depth/size."""
    if depth <= 0:
        # Base case - just return base types
        return rng.choice([("Bool",), ("Int",), ("String",)])
    
    # Randomly choose between base types and function types
    if rng.random() < 0.3:
        # Base type
        return rng.choice([("Bool",), ("Int",), ("String",)])
    else:
        # Function type - recursively generate smaller types for domain and codomain
        # This builds larger types by combining smaller ones
        arg_ty = generate_type(rng, depth // 2)
        ret_ty = generate_type(rng, depth // 2)
        return ("Fun", arg_ty, ret_ty)

def generate_term(rng, depth, target_ty, env):
    """Generate a term of the given type, using variables from env."""
    # Base case - if depth is 0 or target is a base type, return a literal
    if depth <= 0 or target_ty[0] != "Fun":
        return generate_literal(rng, target_ty)
    
    # Otherwise, generate a lambda that matches the target type
    # target_ty is ("Fun", arg_ty, ret_ty)
    arg_ty = target_ty[1]
    ret_ty = target_ty[2]
    
    # Generate a fresh variable name
    var_name = f"x{rng.randrange(1000)}"
    
    # Generate the body with the variable in scope
    body = generate_term(rng, depth - 1, ret_ty, [(var_name, arg_ty)] + env)
    
    return ("Lam", var_name, arg_ty, body)
```

This approach builds function types recursively and generates lambdas that properly consume their argument types, then recursively generates the body with the parameter in scope. Variables get used in the body, and the structure naturally creates more complex terms as depth increases.
