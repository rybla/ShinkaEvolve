```python
def generate_sample(rng: random.Random, depth=10) -> Tuple[Ty, Tm]:
    # Base cases (primitives) - more likely at lower depth
    if depth <= 0:
        i = rng.randrange(0, 3)
        if i == 0:
            return (("Bool",), ("Bool", True))
        elif i == 1:
            return (("Int",), ("Int", 27))
        else:
            return (("String",), ("String", "hello world"))
    
    # Choose which case to generate
    i = rng.randrange(0, 5)  # 5 cases now
    
    if i == 0:
        return (("Bool",), ("Bool", True))
    elif i == 1:
        return (("Int",), ("Int", 27))
    elif i == 2:
        return (("String",), ("String", "hello world"))
    elif i == 3:
        # Lambda case - sometimes use the variable
        ty, tm = generate_sample(rng, depth - 1)
        # Randomly decide whether to use the bound variable
        if rng.random() < 0.5:
            # Use the variable
            return (
                ("Fun", ("Int",), ty),
                ("Lam", "x", ("Int",), ("Var", "x")),
            )
        else:
            # Use the recursive call
            return (
                ("Fun", ("Int",), ty),
                ("Lam", "x", ("Int",), tm),
            )
    else:
        # Application case - apply a lambda to an argument
        func_ty, func_tm = generate_sample(rng, depth - 1)
        arg_ty, arg_tm = generate_sample(rng, depth - 1)
        return (
            ("Fun", (arg_ty, func_ty)),
            ("App", func_tm, arg_tm),
        )
```

This approach adds variety by introducing lambda applications and variable references, which should help with the low scores on variable usage and lambda application metrics.
