def generate_sample(rng: random.Random, depth=10, bound_vars: list = None) -> Tuple[Ty, Tm]:
    if bound_vars is None:
        bound_vars = []
    
    # With low depth, prefer base cases to maintain type correctness
    if depth <= 0 or (not bound_vars and rng.random() < 0.7):
        # Generate base types: Bool, Int, String
        i = rng.randrange(0, 3)
        if i == 0:
            return (("Bool",), ("Bool", rng.choice([True, False])))
        elif i == 1:
            return (("Int",), ("Int", rng.randint(0, 100)))
        else:
            return (("String",), ("String", rng.choice(["a", "b", "c"])))
    
    # When we have bound variables available, occasionally use them
    if bound_vars and rng.random() < 0.3:
        var_name = rng.choice(bound_vars)
        # Return the type and variable reference
        return (("Int",), ("Var", var_name))
    
    # Generate lambda or application based on remaining depth
    i = rng.randrange(0, 2)
    if i == 0 and depth > 1:
        # Create a lambda with a new bound variable
        var = "x"
        ty, tm = generate_sample(rng, depth - 1, bound_vars + [var])
        return (("Fun", ("Int",), ty), ("Lam", var, ("Int",), tm))
    else:
        # Create an application
        func_ty, func_tm = generate_sample(rng, depth - 1, bound_vars)
        arg_ty, arg_tm = generate_sample(rng, depth - 1, bound_vars)
        return (arg_ty, ("App", func_tm, arg_tm))
