def generate_sample(rng: random.Random, depth=10) -> Tuple[Ty, Tm]:
    """
    Generate a sample type and term using a random seed.
    """
    
    if depth <= 0:
        # Base cases - only literals when depth is exhausted
        i = rng.randrange(0, 3)
        if i == 0:
            return (("Bool",), ("Bool", True))
        elif i == 1:
            return (("Int",), ("Int", 27))
        else:
            return (("String",), ("String", "hello world"))
    
    # When depth remains, choose from multiple generation strategies
    i = rng.randrange(0, 6)
    
    if i == 0:
        return (("Bool",), ("Bool", True))
    elif i == 1:
        return (("Int",), ("Int", 27))
    elif i == 2:
        return (("String",), ("String", "hello world"))
    elif i == 3:
        # Lambda that uses the bound variable
        ty, tm = generate_sample(rng, depth - 1)
        return (
            ("Fun", ("Int",), ty),
            ("Lam", "x", ("Int",), ("Var", "x")),
        )
    elif i == 4:
        # Lambda that applies the bound variable to an argument
        arg_ty, arg_tm = generate_sample(rng, depth - 1)
        body_ty, body_tm = generate_sample(rng, depth - 1)
        return (
            ("Fun", arg_ty, body_ty),
            ("Lam", "x", arg_ty, ("App", ("Var", "x"), arg_tm)),
        )
    else:
        # Recursive case - lambda wrapping a recursive call
        ty, tm = generate_sample(rng, depth - 1)
        return (
            ("Fun", ("Int",), ty),
            ("Lam", "x", ("Int",), tm),
        )
