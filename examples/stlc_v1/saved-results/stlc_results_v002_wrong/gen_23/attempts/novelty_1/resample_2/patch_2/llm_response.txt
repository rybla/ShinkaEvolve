```python
def generate_sample(rng: random.Random, depth=10, bound_vars: list = None) -> Tuple[Ty, Tm]:
    if bound_vars is None:
        bound_vars = []
    
    # At low depth or randomly, generate base cases
    if depth <= 0 or rng.random() < 0.3:
        i = rng.randrange(0, 3)
        if i == 0:
            return (("Bool",), ("Bool", rng.choice([True, False])))
        elif i == 1:
            return (("Int",), ("Int", rng.randint(0, 100)))
        else:
            return (("String",), ("String", rng.choice(["a", "b", "c"])))
    
    # When variables are available, occasionally reference them
    if bound_vars and rng.random() < 0.2:
        var_name = rng.choice(bound_vars)
        return (("Int",), ("Var", var_name))
    
    # Choose between creating a lambda or applying one
    if rng.random() < 0.5:
        var = rng.choice(["x", "y", "z"])
        # Create lambda with a new bound variable
        inner_ty, inner_tm = generate_sample(rng, depth - 1, bound_vars + [var])
        return (("Fun", ("Int",), inner_ty), ("Lam", var, ("Int",), inner_tm))
    else:
        # Apply lambda to argument
        func_ty, func_tm = generate_sample(rng, depth - 1, bound_vars)
        arg_ty, arg_tm = generate_sample(rng, depth - 1, bound_vars)
        return (arg_ty, ("App", func_tm, arg_tm))
```

This approach introduces variable references and applications while maintaining type correctness by tracking bound variables and ensuring the generated terms remain well-typed.
