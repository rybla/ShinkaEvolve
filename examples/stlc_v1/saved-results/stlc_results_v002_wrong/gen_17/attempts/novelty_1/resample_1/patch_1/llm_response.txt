def generate_sample(rng: random.Random, depth=10) -> Tuple[Ty, Tm]:
    if depth <= 0:
        # Base case: return simple literals

        if i == 0:
            return (("Bool",), ("Bool", True))
        elif i == 1:
            return (("Int",), ("Int", 27))
        else:
            return (("String",), ("String", "hello world"))

        return (
            ("Fun", arg_ty, ret_ty),

    else:
        # Create App

        arg_ty = ("Int",)
