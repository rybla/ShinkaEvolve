def generate_type(rng: random.Random, size_budget: int) -> Ty:
    """Generate a type with a given size budget."""
    if size_budget <= 1:
        # Base case: generate a base type
        i = rng.randrange(0, 3)
        if i == 0:
            return ("Bool",)
        elif i == 1:
            return ("Int",)
        else:
            return ("String",)
    else:
        # Recursive case: generate a function type
        # Distribute the budget between domain and codomain
        left_budget = rng.randrange(1, size_budget)
        right_budget = size_budget - left_budget
        domain = generate_type(rng, left_budget)
        codomain = generate_type(rng, right_budget)
        return ("Fun", domain, codomain)

def generate_term(rng: random.Random, available_vars: list[str], size_budget: int) -> Tm:

        if available_vars and rng.random() < 0.5:

            if i == 0:

            elif i == 1:
                return ("Int", rng.randint(0, 100))
            else:
                return ("String", rng.choice(["hello", "world", "test"]))
    else:
        # Recursive case: generate a lambda or application
        if rng.random() < 0.5:
            # Generate a lambda
            var_name = f"x{rng.randint(0, 1000)}"
            var_type = generate_type(rng, size_budget // 2)
            body = generate_term(rng, available_vars + [var_name], size_budget // 2)
            return ("Lam", var_name, var_type, body)
        else:
            # Generate an application
            func = generate_term(rng, available_vars, size_budget // 2)
            arg = generate_term(rng, available_vars, size_budget // 2)
            return ("App", func, arg)
