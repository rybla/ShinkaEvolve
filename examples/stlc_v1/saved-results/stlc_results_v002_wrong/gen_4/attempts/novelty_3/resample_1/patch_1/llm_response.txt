def generate_term(rng: random.Random, depth: int, target_ty: Ty, ctx: dict[str, Ty]) -> Tm:

    if depth <= 0 or (isinstance(target_ty, tuple) and target_ty[0] in ("Bool", "Int", "String")):
        return generate_literal(rng, target_ty)
    
    # Occasionally use a variable from context if available
