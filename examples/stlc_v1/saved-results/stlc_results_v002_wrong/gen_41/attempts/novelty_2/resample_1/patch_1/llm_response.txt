def generate_sample(rng: random.Random, depth=10, bound_vars: list[tuple[str, Ty]] = None) -> Tuple[Ty, Tm]:
    if bound_vars is None:
        bound_vars = []
    
    # Base case: when depth is 0, generate simple literals
    if depth <= 0 or (not bound_vars and rng.random() < 0.5):
        i = rng.randrange(0, 3)
        if i == 0:
            return (("Bool",), ("Bool", rng.choice([True, False])))
        elif i == 1:
            return (("Int",), ("Int", rng.randint(0, 100)))
        else:
            return (("String",), ("String", rng.choice(["hello", "world", "test"])))
    
    # At deeper depths, we can create more complex terms
    i = rng.randrange(0, 5 if bound_vars else 4)
    
    if i == 0:
        # Generate a Bool literal
        return (("Bool",), ("Bool", rng.choice([True, False])))
    elif i == 1:
        # Generate an Int literal
        return (("Int",), ("Int", rng.randint(0, 100)))
    elif i == 2:
        # Generate a String literal
        return (("String",), ("String", rng.choice(["hello", "world", "test"])))
    elif i == 3 and bound_vars:
        # Use a variable from bound_vars
        var_name, var_ty = rng.choice(bound_vars)
        return (var_ty, ("Var", var_name))
    else:
        # Create a lambda with a fresh variable
        var_name = f"x{rng.randint(0, 100)}"
        var_ty = rng.choice([("Bool",), ("Int",), ("String",)])
        ret_ty, body = generate_sample(rng, depth - 1, bound_vars + [(var_name, var_ty)])
        return (("Fun", var_ty, ret_ty), ("Lam", var_name, var_ty, body))
