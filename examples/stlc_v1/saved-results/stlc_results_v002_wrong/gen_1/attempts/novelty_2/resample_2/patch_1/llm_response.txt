```python
def generate_sample(rng: random.Random, depth=10) -> Tuple[Ty, Tm]:
    """
    Generate a sample type and term using a random seed.
    """
    # Generate a type first
    ty = generate_type(rng, depth)
    
    # Generate a term with that type
    tm = generate_term(rng, depth, ty, {})
    
    return (ty, tm)

def generate_type(rng: random.Random, depth: int) -> Ty:
    """Generate a type with roughly the given depth."""
    if depth <= 0 or rng.random() < 0.3:
        # Base case: choose a base type
        i = rng.randrange(0, 3)
        if i == 0:
            return ("Bool",)
        elif i == 1:
            return ("Int",)
        else:
            return ("String",)
    else:
        # Recursive case: function type
        arg_ty = generate_type(rng, depth - 1)
        ret_ty = generate_type(rng, depth - 1)
        return ("Fun", arg_ty, ret_ty)

def generate_term(rng: random.Random, depth: int, ty: Ty, context: dict) -> Tm:
    """Generate a term with the given type."""
    if depth <= 0 or (ty[0] == "Bool" and rng.random() < 0.3):
        # Base case: literal or variable
        if rng.random() < 0.3 and context:
            # Use a variable from context
            var_name = rng.choice(list(context.keys()))
            return ("Var", var_name)
        else:
            # Generate a literal based on type
            if ty == ("Bool",):
                return ("Bool", rng.choice([True, False]))
            elif ty == ("Int",):
                return ("Int", rng.randint(0, 100))
            else:
                return ("String", rng.choice(["a", "b", "c"]))
    else:
        # Recursive case: lambda or application
        if rng.random() < 0.5:
            # Lambda
            var_name = "x" + str(depth)
            arg_ty = generate_type(rng, depth - 1)
            body_ty = ty[2]  # Get return type
            body = generate_term(rng, depth - 1, body_ty, {**context, var_name: arg_ty})
            return ("Lam", var_name, arg_ty, body)
        else:
            # Application
            func_ty = ("Fun", generate_type(rng, depth - 1), ty)
            func = generate_term(rng, depth - 1, func_ty, context)
            arg = generate_term(rng, depth - 1, func_ty[1], context)
            return ("App", func, arg)
```

I'm building a recursive type generator that creates function types by combining smaller types, then a term generator that uses variables from context when available. The approach alternates between lambdas and applications to create larger terms while respecting the type constraints.
