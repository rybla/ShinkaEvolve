from typing import Literal, Tuple
import random

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]

type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

# EVOLVE-BLOCK-START

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generates STLC terms using biased higher-order redexes and context-driven type selection.
    """

    def gen_type(nodes: int) -> Ty:
        if nodes <= 1:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        # High bias towards functions (80%) to increase abstraction depth
        if rng.random() < 0.8:
            return ("Fun", gen_type(1), gen_type(nodes - 1))
        return rng.choice([("Bool",), ("Int",), ("String",)])

    def gen_term(ctx: list[Tuple[str, Ty]], target_ty: Ty, budget: int) -> Tm:
        # 1. Variable Lookup (Priority for used_vars metric)
        matches = [n for n, t in ctx if t == target_ty]
        if matches and (budget <= 2 or rng.random() < 0.45):
            return ("Var", rng.choice(matches))

        # 2. Biased Higher-Order Redex Injection
        # Creates ((\v. body) arg) where arg is likely a function
        if budget > 20 and rng.random() < 0.4:
            # 60% chance to force a higher-order argument type
            if rng.random() < 0.6:
                arg_ty = ("Fun", gen_type(1), gen_type(1))
            else:
                arg_ty = rng.choice(ctx)[1] if (ctx and rng.random() < 0.7) else gen_type(1)
            
            vname = f"v{len(ctx)}"
            # Body gets the lion's share of budget
            lam_body = gen_term(ctx + [(vname, arg_ty)], target_ty, int(budget * 0.6))
            lam_node = ("Lam", vname, arg_ty, lam_body)
            arg_node = gen_term(ctx, arg_ty, int(budget * 0.25))
            return ("App", lam_node, arg_node)

        # 3. Deep Callable Search
        if budget > 5 and ctx:
            potential = []
            for name, ty in ctx:
                if ty[0] == "Fun" and ty[2] == target_ty:
                    potential.append((name, ty[1]))
            if potential and rng.random() < 0.5:
                vname, arg_ty = rng.choice(potential)
                return ("App", ("Var", vname), gen_term(ctx, arg_ty, budget - 2))

        # 4. Mandatory Abstraction for Function Types
        if target_ty[0] == "Fun":
            _, arg_ty, ret_ty = target_ty
            vname = f"v{len(ctx)}"
            return ("Lam", vname, arg_ty, gen_term(ctx + [(vname, arg_ty)], ret_ty, budget - 1))

        # 5. General Application with Context Bias
        if budget > 3:
            # Prefer types already present in context to increase Var usage
            if ctx and rng.random() < 0.85:
                arg_ty = rng.choice(ctx)[1]
            else:
                # 50% chance for higher-order application
                arg_ty = ("Fun", gen_type(1), gen_type(1)) if rng.random() < 0.5 else gen_type(1)
            
            fun_ty = ("Fun", arg_ty, target_ty)
            return ("App", 
                    gen_term(ctx, fun_ty, int(budget * 0.7)), 
                    gen_term(ctx, arg_ty, int(budget * 0.2)))

        # 6. Leaf Fallbacks
        if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
        if target_ty == ("Int",): return ("Int", rng.randint(1, 100))
        if target_ty == ("String",): return ("String", "s")
        return ("Int", 42)

    # Target type size 8-11 nodes
    target_type = gen_type(rng.randint(8, 11))
    # Initial budget 115 to target ~25-30 nodes total
    term = gen_term([], target_type, 115)
    return target_type, term
