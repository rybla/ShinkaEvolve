from typing import Literal, Tuple
import random
import tqdm

"""
`Ty` and `Tm` define the shape of a simply-typed lambda-calculus deeply embedded in Python.
"""

type Ty = BoolTy | IntTy | StringTy | FunTy
type BoolTy = Tuple[Literal["Bool"]]
type IntTy = Tuple[Literal["Int"]]
type StringTy = Tuple[Literal["String"]]
type FunTy = Tuple[Literal["Fun"], Ty, Ty]


type Tm = LitTm | VarTm | LamTm | AppTm
type LitTm = (
    Tuple[Literal["Bool"], bool]
    | Tuple[Literal["Int"], int]
    | Tuple[Literal["String"], str]
)
type VarTm = Tuple[Literal["Var"], str]
type LamTm = Tuple[Literal["Lam"], str, Ty, Tm]
type AppTm = Tuple[Literal["App"], Tm, Tm]

def generate_sample(rng: random.Random) -> Tuple[Ty, Tm]:
    """
    Generate a sample type and term using a random seed with tiered type-driven synthesis.
    """

    def gen_type(max_nodes: int) -> Ty:
        if max_nodes <= 1:
            return rng.choice([("Bool",), ("Int",), ("String",)])
        if rng.random() < 0.75:  # High probability of function types for depth
            left_size = rng.randint(1, max_nodes - 1)
            right_size = max_nodes - 1 - left_size
            return ("Fun", gen_type(left_size), gen_type(right_size))
        return rng.choice([("Bool",), ("Int",), ("String",)])

    def gen_term(ctx: list[Tuple[str, Ty]], target_ty: Ty, budget: int) -> Tm:
        # Tier 1: Variable Match (High Priority for used_vars metric)
        matching_vars = [name for name, ty in ctx if ty == target_ty]
        if matching_vars and (budget <= 2 or rng.random() < 0.45):
            return ("Var", rng.choice(matching_vars))

        # Tier 2: Callable Context Lookup
        if budget > 5:
            callables = [(n, t[1]) for n, t in ctx if t[0] == "Fun" and t[2] == target_ty]
            if callables and rng.random() < 0.6:
                vname, arg_ty = rng.choice(callables)
                return ("App", ("Var", vname), gen_term(ctx, arg_ty, budget - 2))

        # Tier 3: Redex-Chain Injection (The "Power Move")
        if budget > 30 and rng.random() < 0.35:
            # Pick arg types from context to ensure the App arguments can be Vars
            a1_ty = rng.choice(ctx)[1] if (ctx and rng.random() < 0.7) else gen_type(1)
            a2_ty = rng.choice(ctx)[1] if (ctx and rng.random() < 0.7) else gen_type(1)
            v1, v2 = f"v{len(ctx)}", f"v{len(ctx)+1}"
            
            body = gen_term(ctx + [(v1, a1_ty), (v2, a2_ty)], target_ty, int(budget * 0.55))
            inner_lam = ("Lam", v2, a2_ty, body)
            outer_lam = ("Lam", v1, a1_ty, inner_lam)
            
            arg1 = gen_term(ctx, a1_ty, budget // 6)
            arg2 = gen_term(ctx, a2_ty, budget // 6)
            return ("App", ("App", outer_lam, arg1), arg2)

        # Tier 4: Explicit Redex Injection
        if budget > 12 and rng.random() < 0.4:
            arg_ty = rng.choice(ctx)[1] if (ctx and rng.random() < 0.7) else gen_type(2)
            vname = f"v{len(ctx)}"
            lam = ("Lam", vname, arg_ty, gen_term(ctx + [(vname, arg_ty)], target_ty, int(budget * 0.6)))
            arg = gen_term(ctx, arg_ty, int(budget * 0.2))
            return ("App", lam, arg)

        # Tier 5: Mandatory Lambda for Fun types
        if target_ty[0] == "Fun":
            _, arg_ty, ret_ty = target_ty
            var_name = f"v{len(ctx)}"
            return ("Lam", var_name, arg_ty, gen_term(ctx + [(var_name, arg_ty)], ret_ty, budget - 1))

        # Tier 6: General Application (Split budget to favor function side)
        if budget > 3:
            if ctx and rng.random() < 0.8:
                arg_ty = rng.choice(ctx)[1]
            else:
                arg_ty = gen_type(rng.randint(1, 2))
            
            fun_ty = ("Fun", arg_ty, target_ty)
            return ("App", gen_term(ctx, fun_ty, int(budget * 0.7)), gen_term(ctx, arg_ty, int(budget * 0.2)))

        # Safety Fallbacks
        if target_ty == ("Bool",): return ("Bool", rng.choice([True, False]))
        if target_ty == ("Int",): return ("Int", rng.randint(0, 500))
        if target_ty == ("String",): return ("String", "s")

        return ("Lam", "unused", ("Int",), ("Int", 0))

    target_type = gen_type(rng.randint(8, 11))
    # High initial budget to maximize term size metrics
    generated_term = gen_term([], target_type, 120)
    return target_type, generated_term

import lc


def run() -> lc.RunOutput:
    """Run the analysis"""

    size = 1000
    rng = random.Random()

    return lc.run(
        size=size,
        rng=rng,
        generate_sample=generate_sample,
    )
